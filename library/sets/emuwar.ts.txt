/** a 3D Lewis gun, barrel along -z, receiver at the origin (bipod feet ~0.27 m below) */
export const lewisGun = () => {
  const g = new THREE.Group();
  const steel = phong("ewSteel", { color: "#2c2d34", specular: "#a8b0c0", shininess: 60 }), steel2 = phong("ewSteel2", { color: "#3d3f48", specular: "#c0c8d8", shininess: 80 });
  const woodM = phong("ewStock", { color: "#9a6236", specular: "#e0b080", shininess: 30 });
  const cyl = (r0: number, r1: number, z0: number, z1: number, m: THREE.Material, y = 0) => { const c = new THREE.Mesh(new THREE.CylinderGeometry(r0, r1, z1 - z0, 20).rotateX(Math.PI / 2).translate(0, y, (z0 + z1) / 2), m); c.castShadow = true; g.add(c); return c; };
  cyl(0.066, 0.066, -0.92, -0.14, steel);                     // the fat cooling shroud
  for (let i = 0; i < 7; i++) cyl(0.071, 0.071, -0.86 + i * 0.1, -0.84 + i * 0.1, steel2); // shroud bands
  cyl(0.045, 0.066, -1.0, -0.92, steel);                      // tapering mouth
  cyl(0.016, 0.016, -1.08, -1.0, steel2);                     // muzzle
  const rec = new THREE.Mesh(new THREE.BoxGeometry(0.1, 0.12, 0.36).translate(0, 0, 0.04), steel); rec.castShadow = true; g.add(rec);
  const pan = new THREE.Mesh(new THREE.CylinderGeometry(0.16, 0.16, 0.05, 32).translate(0, 0.1, -0.02), steel2); pan.castShadow = true; g.add(pan);
  const hub = new THREE.Mesh(new THREE.CylinderGeometry(0.035, 0.035, 0.07, 12).translate(0, 0.12, -0.02), woodM); g.add(hub);
  for (let i = 0; i < 12; i++) { const a = (i / 12) * Math.PI * 2, s = new THREE.Mesh(new THREE.BoxGeometry(0.012, 0.055, 0.11), steel); s.position.set(Math.cos(a) * 0.1, 0.1, -0.02 + Math.sin(a) * 0.1); s.rotation.y = -a; g.add(s); }

  const grip = new THREE.Mesh(new THREE.BoxGeometry(0.05, 0.13, 0.06).translate(0, -0.1, 0.17), woodM); grip.rotation.x = -0.3; g.add(grip);
