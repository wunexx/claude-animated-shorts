export type DzScheme = "grey" | "dazzle" | "half" | "false";
const DZ_INK = "#15171c", DZ_NAVY = "#25375f", DZ_BLUE = "#6f8fb8", DZ_PALE = "#eef0ea", DZ_GREY = "#8b939b";
const WL = 0.235; // waterline as a fraction of hull height from the keel (hull y -2 .. 6.5)
/** bold slanted dazzle bands + wedges across a w x h canvas region [x0, x1) */
const dazzlePaint = (g: CanvasRenderingContext2D, x0: number, x1: number, h: number, seed: number) => {
  const r = rng(seed), cols = [DZ_INK, DZ_NAVY, DZ_BLUE, DZ_INK, "#a9b3bd", DZ_NAVY];
  g.fillStyle = DZ_PALE; g.fillRect(x0, 0, x1 - x0, h);
  let x = x0 - h * 0.6, i = 0;
  while (x < x1 + h) {
    const w = h * (0.18 + r() * 0.32), sk = h * (r() < 0.5 ? 0.55 : -0.45) * (0.6 + r() * 0.6);
    g.fillStyle = cols[i++ % cols.length];
    g.beginPath(); g.moveTo(x, 0); g.lineTo(x + w, 0); g.lineTo(x + w + sk, h); g.lineTo(x + sk, h); g.closePath(); g.fill();
    if (r() < 0.55) { // a wedge breaking the band
      g.fillStyle = r() < 0.5 ? DZ_PALE : DZ_BLUE;
      const y = h * (0.15 + r() * 0.5);
      g.beginPath(); g.moveTo(x + w * 0.2, y); g.lineTo(x + w * 1.8, y + h * 0.12); g.lineTo(x + w * 0.4, y + h * 0.32); g.closePath(); g.fill();
    }
    x += w * (1.5 + r() * 0.9);
  }
};
const greyPaint = (g: CanvasRenderingContext2D, x0: number, x1: number, h: number) => {
  g.fillStyle = DZ_GREY; g.fillRect(x0, 0, x1 - x0, h);
  g.strokeStyle = "rgba(40,44,52,0.18)"; g.lineWidth = 2;
  for (let x = x0 + 40; x < x1; x += 96) { g.beginPath(); g.moveTo(x, 0); g.lineTo(x, h); g.stroke(); } // plate seams
  g.fillStyle = "rgba(255,255,255,0.08)"; g.fillRect(x0, 0, x1 - x0, h * 0.08);
};
/** hull side texture, u along the ship (0 stern .. 1 bow), v keel .. deck */
export const hullTex = (s: DzScheme) => ctex("dzHull_" + s, 2048, 512, (g) => {
  const W = 2048, H = 512, wl = H * (1 - WL);
  if (s === "grey") greyPaint(g, 0, W, wl);
  else if (s === "half") {
    dazzlePaint(g, 0, W, wl, 7);
    g.save(); g.beginPath(); // ragged paint edge at u ~0.475: the bow half still grey
    g.moveTo(W * 0.475, 0); for (let y = 0; y <= wl; y += 32) g.lineTo(W * (0.475 + 0.006 * Math.sin(y * 0.37)), y);
    g.lineTo(W, wl); g.lineTo(W, 0); g.closePath(); g.clip(); greyPaint(g, 0, W, wl); g.restore();
  } else dazzlePaint(g, 0, W, wl, s === "false" ? 11 : 7);
  if (s === "false") { // a painted FALSE bow wave near the stern + a dark broken "stern" at the real bow
    g.fillStyle = DZ_INK; g.beginPath(); g.moveTo(0, wl * 0.15); g.lineTo(W * 0.2, wl * 0.05); g.lineTo(W * 0.24, wl); g.lineTo(0, wl); g.closePath(); g.fill();
    g.fillStyle = "#ffffff"; g.beginPath(); g.moveTo(W * 0.02, wl); g.quadraticCurveTo(W * 0.1, wl * 0.45, W * 0.19, wl * 0.62);
    g.quadraticCurveTo(W * 0.12, wl * 0.8, W * 0.16, wl); g.closePath(); g.fill();
    g.fillStyle = DZ_PALE; g.beginPath(); g.moveTo(W * 0.86, 0); g.lineTo(W, 0); g.lineTo(W, wl); g.lineTo(W * 0.93, wl); g.closePath(); g.fill();
  }
  g.fillStyle = "#1c1c20"; g.fillRect(0, wl - 10, W, 12);           // black boot line
  g.fillStyle = "#9c3a2c"; g.fillRect(0, wl + 2, W, H - wl);         // red antifouling below the waterline
});
const superTex = (s: DzScheme) => ctex("dzSuper_" + s, 512, 256, (g) => {
  if (s === "grey" || s === "half") greyPaint(g, 0, 512, 256); else dazzlePaint(g, 0, 512, 256, s === "false" ? 23 : 19);
  g.fillStyle = "#2a2f3a"; for (let x = 30; x < 500; x += 56) { g.beginPath(); g.arc(x, 110, 11, 0, 7); g.fill(); } // portholes
});
const funnelTex = (s: DzScheme) => ctex("dzFun_" + s, 256, 512, (g) => {
  if (s === "grey" || s === "half") greyPaint(g, 0, 256, 512); else dazzlePaint(g, 0, 256, 512, s === "false" ? 31 : 29);
  g.fillStyle = "#16161a"; g.fillRect(0, 0, 256, 70);
});
/** a 1917 cargo steamer, bow toward local +x, waterline at y 0. set(scheme) swaps the paint (all schemes prebuilt). */
export const steamer = (L = 40, B = 9) => {
  const grp = new THREE.Group(), y0 = -2, deck = 6.5, sx = -L / 2 + 3.2, bowS = L / 2 - 0.22 * L;
  const zOf = (x: number) => {
    if (x < sx) return (B / 2) * Math.sqrt(Math.max(0, 1 - ((sx - x) / 3.2) ** 2));
    if (x > bowS) return (B / 2) * Math.max(0, (L / 2 - x) / (L / 2 - bowS)) ** 0.75;
    return B / 2;
  };
  const yTop = (x: number) => deck + 1.3 * Math.max(0, (x - L / 4) / (L / 4)) ** 2;
  const N = 44, xs = Array.from({ length: N + 1 }, (_, i) => -L / 2 + (L * i) / N);
  const wall: THREE.BufferGeometry[] = [], top: THREE.BufferGeometry[] = [];
  for (let i = 0; i < N; i++) {
    const xa = xs[i], xb = xs[i + 1], za = zOf(xa), zb = zOf(xb), ua = i / N, ub = (i + 1) / N;
    wall.push(quad(V(xa, y0, za), V(xb, y0, zb), V(xb, yTop(xb), zb), V(xa, yTop(xa), za), ub, 1, ua, 0));
    wall.push(quad(V(xb, y0, -zb), V(xa, y0, -za), V(xa, yTop(xa), -za), V(xb, yTop(xb), -zb), ua, 1, ub, 0));
    top.push(quad(V(xa, yTop(xa), za), V(xb, yTop(xb), zb), V(xb, yTop(xb), -zb), V(xa, yTop(xa), -za)));
  }
  const wallGeo = mergeGeometries(wall, false), topGeo = mergeGeometries(top, false);
  const deckMat = lam("dzDeck", () => ({ color: "#9a7655" }));
  const steel = lam("dzSteel", () => ({ color: "#5d646d" })), dark = lam("dzDark", () => ({ color: "#2b2f37" }));
  const glass = lam("dzWin", () => ({ color: "#1f2a38", emissive: "#2a3a50" }));
  const schemes: Record<string, THREE.Group> = {};
  for (const s of ["grey", "dazzle", "half", "false"] as DzScheme[]) {
    const sg = new THREE.Group(), b = new Batch();
    b.add(wallGeo, lam("dzHullM_" + s, () => ({ map: hullTex(s) })));
    const sup = lam("dzSupM_" + s, () => ({ map: superTex(s) }));
    b.add(boxGeo(V(-L * 0.13, deck, -B * 0.33), V(L * 0.1, deck + 2.8, B * 0.33)), sup);       // deck house
    b.add(boxGeo(V(-L * 0.02, deck + 2.8, -B * 0.29), V(L * 0.08, deck + 4.7, B * 0.29)), sup);  // bridge
    b.add(boxGeo(V(L * 0.04, deck + 4.7, -B * 0.31), V(L * 0.085, deck + 5.0, B * 0.31)), dark);  // bridge roof
    b.add(new THREE.CylinderGeometry(L * 0.034, L * 0.036, 6.4, 20).translate(-L * 0.075, deck + 2.8 + 3.2, 0), lam("dzFunM_" + s, () => ({ map: funnelTex(s) })));
    b.build(sg); grp.add(sg); schemes[s] = sg;
  }
  const b = new Batch();
  b.add(topGeo, deckMat);
  b.add(boxGeo(V(L * 0.085 - 0.05, deck + 3.3, -B * 0.28), V(L * 0.085 + 0.02, deck + 4.2, B * 0.28)), glass); // bridge windows
  for (const mx of [L * 0.3, -L * 0.33]) {
    b.add(new THREE.CylinderGeometry(0.16, 0.22, 14, 8).translate(mx, deck + 7, 0), steel);                   // mast
    b.add(new THREE.CylinderGeometry(0.1, 0.1, B * 0.8, 6).rotateX(Math.PI / 2).translate(mx, deck + 10.5, 0), steel); // yard
    b.add(boxGeo(V(mx - 2.2, deck, -1.6), V(mx + 2.2, deck + 1.1, 1.6)), dark);                              // hatch
  }
  b.build(grp);
  const set = (s: DzScheme) => { for (const k in schemes) schemes[k].visible = k === s; };
  set("grey");
  return { g: grp, set };
};

/** a long-handled roller for the hook painter: from the hands to the hull (both world points) */
export const roller = () => {
  const grp = new THREE.Group();
  const handle = new THREE.Mesh(new THREE.CylinderGeometry(0.025, 0.025, 1, 8), lam("dzHandle", () => ({ color: "#c09060" })));
  const head = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.09, 0.42, 16).rotateZ(Math.PI / 2), lam("dzRollerHead", () => ({ color: "#1b1d22" })));
  grp.add(handle, head);
  const a = new THREE.Vector3(), c = new THREE.Vector3();
  const pose = (hand: [number, number, number], tip: [number, number, number]) => {
    a.set(...hand); c.set(...tip);
    handle.position.copy(a).add(c).multiplyScalar(0.5); handle.scale.set(1, a.distanceTo(c), 1);
    handle.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), c.clone().sub(a).normalize());
    head.position.copy(c);
  };
  return { g: grp, pose };
};

  for (const [x, k] of [[-3.1, 0], [-0.9, 1], [3.2, 2]] as number[][])  // dazzle design boards on the back wall

  const T = STORY.TABLE, wood = lam("dzTable", () => ({ color: "#6e4b34" }));
  b.add(boxGeo(V(T[0] - 0.85, T[1] - 0.06, T[2] - 0.6), V(T[0] + 0.85, T[1], T[2] + 0.6)), wood);

  const S = STORY.SCOPE, brass = lam("dzBrassScope", () => ({ color: "#c08a26", emissive: "#6a4208" })), rubber = lam("dzRubber", () => ({ color: "#15161a" }));

  for (const a of [0, 2.1, 4.2]) b.add(new THREE.CylinderGeometry(0.02, 0.02, 0.9, 6).rotateZ(0.32).rotateY(a).translate(S[0] + 0.14 * Math.cos(a), 0.42, S[2] - 0.14 * Math.sin(a)), lam("dzTripod", () => ({ color: "#3a2a20" })));
  // a shelf of finished ship models on the left wall
  b.add(boxGeo(V(-4.9, 1.45, z0 - 3.8), V(-4.5, 1.5, z0 - 1.6)), wood);
