const lathe = (pts: [number, number][], seg = 40, p0 = 0, pl = Math.PI * 2) => new THREE.LatheGeometry(pts.map(([r, y]) => new THREE.Vector2(r, y)), seg, p0, pl);
const M = (g: THREE.BufferGeometry, m: THREE.Material, x = 0, y = 0, z = 0) => { const o = new THREE.Mesh(g, m); o.position.set(x, y, z); o.castShadow = o.receiveShadow = true; return o; };
const ink = (o: THREE.Mesh, w = 0.012) => { const h = inkHull(o, w); o.parent?.add(h); return h; };
const addInk = (g: THREE.Object3D, m: THREE.Mesh, w = 0.012) => { g.add(m); g.add(inkHull(m, w)); return m; };

// ---------------- the lens ----------------
export const LENS_R = 0.9, LENS_H = 2.3, LENS_Y = 0.99, BULL_Y = 1.15; // bullseye centre height above the lens base
/** a first-order Fresnel lens: 8 bullseye panels (glowing centres + concentric prism rings), catadioptric prism rings
 *  above and below, brass frames; `bulls` = centre discs (colour set per light), `beams` = 8 soft additive cones */
export const makeLens = (beamLen = 40, beamR = 1.4) => {
  const g = new THREE.Group(), glass = glazeM("#bfeff0"), glass2 = glazeM("#e8fbfb"), brass = metalM("#d9a030");
  const bullMat = new THREE.MeshBasicMaterial({ color: new THREE.Color(0.8, 0.95, 0.95), fog: false });
  const bulls: THREE.Mesh[] = [];
  const beamMat = new THREE.MeshBasicMaterial({ color: new THREE.Color(1.0, 0.85, 0.5), transparent: true, opacity: 0.15, depthWrite: false, blending: THREE.AdditiveBlending, side: THREE.DoubleSide, fog: false });
  const beams = new THREE.Group();
  const ap = LENS_R * Math.cos(Math.PI / 8), pw = 2 * LENS_R * Math.sin(Math.PI / 8);
  for (let i = 0; i < 8; i++) {
    const a = (i / 8) * Math.PI * 2, pg = new THREE.Group();
    // panel back (glass), rings, glowing centre
    pg.add(M(new THREE.PlaneGeometry(pw * 0.98, 1.0), glass, 0, BULL_Y, ap - 0.01));
    for (let k = 0; k < 5; k++) { const r = 0.11 + k * 0.075; pg.add(M(new THREE.TorusGeometry(r, 0.024, 6, 36), k % 2 ? glass : glass2, 0, BULL_Y, ap + 0.01)); }
    const bull = M(new THREE.CircleGeometry(0.1, 28), bullMat, 0, BULL_Y, ap + 0.02); pg.add(bull); bulls.push(bull);
    pg.add(M(new THREE.CylinderGeometry(0.026, 0.026, LENS_H, 8), brass, pw / 2, LENS_H / 2, ap));       // vertical astragal
    for (const y of [0.64, 1.66]) pg.add(M(new THREE.BoxGeometry(pw, 0.05, 0.05), brass, 0, y, ap));
    // beam: a long soft cone out of the bullseye
    const cone = new THREE.Mesh(new THREE.ConeGeometry(beamR, beamLen, 24, 1, true).translate(0, -beamLen / 2, 0).rotateX(-Math.PI / 2), beamMat);
    cone.position.set(0, BULL_Y, ap); const bg = new THREE.Group(); bg.add(cone); bg.rotation.y = a; beams.add(bg);
    pg.rotation.y = a; g.add(pg);
  }
  // catadioptric prism rings above and below (a beehive), brass rings between the sections
  const ringsAt = (y0: number, y1: number, up: boolean) => {
    for (let k = 0; k < 9; k++) {
      const u = (k + 0.5) / 9, y = y0 + (y1 - y0) * u, r = LENS_R * (up ? Math.cos(u * 1.05) : Math.cos((1 - u) * 0.95)) * 0.98;
      g.add(M(new THREE.TorusGeometry(r, 0.034, 6, 48).rotateX(Math.PI / 2), k % 2 ? glass : glass2, 0, y, 0));
    }
    const prof: [number, number][] = [];
    for (let k = 0; k <= 10; k++) { const u = k / 10, y = y0 + (y1 - y0) * u; prof.push([LENS_R * (up ? Math.cos(u * 1.05) : Math.cos((1 - u) * 0.95)) * 0.94, y]); }
    g.add(M(lathe(prof, 32), glass));
  };
  ringsAt(1.66, LENS_H, true); ringsAt(0.02, 0.64, false);
  for (const y of [0.02, 0.64, 1.66]) g.add(M(new THREE.TorusGeometry(LENS_R * (y < 1 ? (y < 0.3 ? 0.56 : 0.99) : 0.99), 0.04, 8, 48).rotateX(Math.PI / 2), brass, 0, y, 0));
  const cap = M(new THREE.CylinderGeometry(0.42, 0.5, 0.08, 24), brass, 0, LENS_H + 0.02, 0); g.add(cap);
  g.add(M(new THREE.SphereGeometry(0.09, 16, 12), brass, 0, LENS_H + 0.14, 0));
  const plate = M(new THREE.CylinderGeometry(0.95, 0.95, 0.06, 40), brass, 0, -0.02, 0); g.add(plate);
  return { g, bulls, bullMat, beams, beamMat };
};

const gear = (r: number, n: number, th: number, m: THREE.Material) => {
  const g = new THREE.Group();
  g.add(M(new THREE.CylinderGeometry(r, r, th, 32).rotateX(Math.PI / 2), m));
  for (let i = 0; i < n; i++) { const a = (i / n) * Math.PI * 2, t = M(new THREE.BoxGeometry(r * 0.22, r * 0.2, th), m, Math.cos(a) * (r + r * 0.08), Math.sin(a) * (r + r * 0.08), 0); t.rotation.z = a; g.add(t); }
  g.add(M(new THREE.CylinderGeometry(r * 0.2, r * 0.2, th * 1.6, 12).rotateX(Math.PI / 2), metalM("#3a3a44")));
  return g;
};

/** the brass clockwork (front of the pedestal): frame, 4 gears, a cable drum, a fly governor. Returns spin(t, k). */
const makeClock = () => {
  const g = new THREE.Group(), brass = metalM("#c9962e"), iron = metalM("#3a3d48");
  for (const x of [-0.25, 0.25]) for (const z of [-0.13, 0.13]) addInk(g, M(new THREE.CylinderGeometry(0.018, 0.018, 0.56, 8), brass, x, 0.28, z), 0.008);
  for (const y of [0.0, 0.56]) addInk(g, M(new THREE.BoxGeometry(0.56, 0.035, 0.3), brass, 0, y, 0), 0.008);
  const gs = [gear(0.15, 18, 0.025, brass), gear(0.075, 10, 0.03, brass), gear(0.1, 14, 0.025, brass), gear(0.05, 8, 0.03, brass)];
  const at: [number, number, number][] = [[-0.08, 0.32, 0.14], [0.13, 0.4, 0.15], [0.12, 0.17, 0.12], [-0.12, 0.12, 0.155]];
  gs.forEach((q, i) => { q.position.set(...at[i]); g.add(q); });
  g.add(M(new THREE.PlaneGeometry(0.5, 0.52), lam("lh_clockBack", () => ({ color: "#efe3c4", side: THREE.DoubleSide })), 0, 0.28, -0.1));
  const drum = M(new THREE.CylinderGeometry(0.075, 0.075, 0.34, 20).rotateZ(Math.PI / 2), lam("lh_drum", () => ({ map: rep(stripesT("#8a6a4a", "#6a4a32", 16), 4, 1) })), 0, 0.2, -0.02); g.add(drum);
  // governor: a vertical spindle with two vanes above the frame
  const gov = new THREE.Group(); gov.position.set(0, 0.58, 0);
  addInk(gov, M(new THREE.CylinderGeometry(0.012, 0.012, 0.32, 8), iron, 0, 0.16, 0), 0.006);
  for (const s of [-1, 1]) addInk(gov, M(new THREE.BoxGeometry(0.16, 0.08, 0.008), brass, s * 0.09, 0.26, 0), 0.006);
  g.add(gov);
  const spin = (t: number, k: number) => {
    const dir = [1, -2, -1.5, 3];
    gs.forEach((q, i) => { q.rotation.z = t * k * dir[i]; });
    drum.rotation.x = t * k * 0.3; gov.rotation.y = t * k * 9;
  };
  return { g, spin };
};

/** the pedestal: cast-iron column + one of three tops (old wheels / closed float / cutaway float). Returns per-era groups. */

  const g = new THREE.Group();

  const section = (pts: [number, number][], col: string, key: string, off = 0.001) => {
