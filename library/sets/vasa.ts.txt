/** half-beam of the hull at station x (-20 stern .. 20 bow) and height y (keel -4.5 .. top) */
export const hullHalf = (x: number, y: number) => {
  const u = Math.max(-1, Math.min(1, x / 20));
  const len = u > 0 ? Math.pow(Math.max(0, 1 - Math.pow(u, 2.4)), 0.55) : 0.7 + 0.3 * Math.pow(Math.max(0, 1 - Math.pow(-u, 3)), 0.5);
  const prof = y < 1 ? Math.pow(Math.sin(Math.min(1, Math.max(0, (y + 4.5) / 5.5)) * Math.PI / 2), 0.75) * 0.97 + 0.03 : 1 - 0.2 * Math.pow((y - 1) / 6.6, 1.4);
  return 5 * len * Math.max(0.03, prof);
};

/** a thin round line from a to b (rigging, cables) */
const rope = (b: Batch, mat: THREE.Material, a: THREE.Vector3, c: THREE.Vector3, r = 0.05) => {
  const d = c.clone().sub(a), L = d.length();
  const geo = new THREE.CylinderGeometry(r, r, L, 5, 1, true);
  const q = new THREE.Quaternion().setFromUnitVectors(V(0, 1, 0), d.normalize());
  b.add(geo, mat, new THREE.Matrix4().compose(a.clone().add(c).multiplyScalar(0.5), q, V(1, 1, 1)));
};
export type Ship = { g: THREE.Group; rig: THREE.Group; flags: THREE.Group; cables: THREE.Group; lids: { o: THREE.Object3D; s: number; k: number }[]; guns: THREE.Object3D[]; sails: THREE.Mesh[]; furls: THREE.Object3D[]; ports: THREE.Vector3[] };
export const PORT_ROWS = [2.2, 4.6];
export const PORT_XS = [-9.6, -7.2, -4.8, -2.4, 0, 2.4, 4.8, 7.2, 9.6, 12];
/** builds the warship into a new group (local frame: bow +x, waterline y 0, deck y 6.9) */
export const buildShip = (parent: THREE.Object3D): Ship => {
  const g = new THREE.Group(), rig = new THREE.Group(), flags = new THREE.Group(), cables = new THREE.Group();
  g.add(rig); g.add(flags); g.add(cables); parent.add(g);
  const b = new Batch(), rb = new Batch(), fb = new Batch(), cb = new Batch();
  const hullM = lam("vasaHullM", () => ({ map: hullTex(), side: THREE.DoubleSide }));
  // ---- lofted hull sides ----
  const NX = 48, NY = 22;
  for (const s of [-1, 1]) {
    const pos: number[] = [], uv: number[] = [], idx: number[] = [];
    for (let i = 0; i <= NX; i++) {
      const x = -20 + (40 * i) / NX, top = hullTop(x), bot = -4.5 + 1.2 * STORY.smooth((x - 15) / 5);
      for (let j = 0; j <= NY; j++) {
        const y = bot + ((top - bot) * j) / NY;
        pos.push(x, y, s * hullHalf(x, y)); uv.push(i / NX, (y + 4.5) / 16.5);
      }
    }
    for (let i = 0; i < NX; i++) for (let j = 0; j < NY; j++) {
      const a = i * (NY + 1) + j, c = a + NY + 1;
      if (s > 0) idx.push(a, c, a + 1, a + 1, c, c + 1); else idx.push(a, a + 1, c, a + 1, c + 1, c);
    }
    const geo = new THREE.BufferGeometry();
    geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3)); geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
    geo.setIndex(idx); geo.computeVertexNormals();
    b.add(geo, hullM);
  }
  // ---- transom (flat stern) ----
  {
    const pts: THREE.Vector2[] = [];
    const top = hullTop(-20);
    for (let j = 0; j <= 16; j++) { const y = -4.5 + ((top + 4.5) * j) / 16; pts.push(new THREE.Vector2(hullHalf(-20, y), y)); }
    for (let j = 16; j >= 0; j--) { const y = -4.5 + ((top + 4.5) * j) / 16; pts.push(new THREE.Vector2(-hullHalf(-20, y), y)); }
    const sh = new THREE.Shape(pts), geo = new THREE.ShapeGeometry(sh);
    const p = geo.attributes.position, uvA = geo.attributes.uv;
    for (let k = 0; k < p.count; k++) { const z = p.getX(k), y = p.getY(k); uvA.setXY(k, 0.5 + z / 7, (y - 4) / (top - 4)); }
    geo.rotateY(-Math.PI / 2).translate(-20, 0, 0);
    b.add(geo, lam("vasaTransomM", () => ({ map: transomTex(), side: THREE.DoubleSide })));
  }
  // ---- decks + bulkheads ----
  const deckM = lam("vasaDeckM", () => ({ map: rep(deckTex(), 6, 2) }));
  const deckStrip = (x0: number, x1: number, y: number) => {
    const pos: number[] = [], uv: number[] = [], idx: number[] = [], n = 20;
    for (let i = 0; i <= n; i++) { const x = x0 + ((x1 - x0) * i) / n, w = hullHalf(x, y) * 0.99; pos.push(x, y, -w, x, y, w); uv.push(i / n, 0, i / n, 1); }
    for (let i = 0; i < n; i++) { const a = 2 * i; idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2); }
    const geo = new THREE.BufferGeometry(); geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3)); geo.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2)); geo.setIndex(idx); geo.computeVertexNormals();
    b.add(geo, deckM);
  };
  deckStrip(-12, 14, DECK_Y); deckStrip(-19.9, -12, POOP_Y); deckStrip(14, 19.2, FORE_Y);
  const bulk = (x: number, y0: number, y1: number) => {
    const w = hullHalf(x, (y0 + y1) / 2) * 0.98;
    b.add(new THREE.PlaneGeometry(2 * w, y1 - y0).rotateY(Math.PI / 2).translate(x, (y0 + y1) / 2, 0), lam("vasaBulk", () => ({ color: "#6b4528", side: THREE.DoubleSide })));
    for (let k = -2; k <= 2; k++) b.add(boxGeo(V(x - 0.02, y0 + 0.9, k * 1.6 - 0.35), V(x + 0.04, y0 + 2.0, k * 1.6 + 0.35)), lam("vasaDoor", () => ({ color: "#2e1c12" })));
    b.add(boxGeo(V(x - 0.15, y1 - 0.15, -w), V(x + 0.15, y1 + 0.85, w)), lam("vasaRailGold", () => ({ color: "#b8862e" })));
  };
  bulk(-12, DECK_Y, POOP_Y); bulk(14, DECK_Y, FORE_Y);
  // gold rail caps along the hull top
  for (const s of [-1, 1]) for (let i = 0; i < 40; i++) {
    const x0 = -20 + i, x1 = x0 + 1, t0 = hullTop(x0), t1 = hullTop(x1);
    const a = V(x0, t0, s * hullHalf(x0, t0)), c = V(x1, t1, s * hullHalf(x1, t1));
    rope(b, lam("vasaRailGold", () => ({ color: "#b8862e" })), a, c, 0.16);
  }
  // keel + stem + beakhead
  b.add(boxGeo(V(-20, -4.8, -0.25), V(17, -4.3, 0.25)), lam("vasaKeel", () => ({ color: "#3a2416" })));
  b.add(new THREE.CylinderGeometry(0.5, 0.9, 6, 8).rotateZ(-1.1).translate(22, 9.6, 0), lam("vasaBeak", () => ({ color: "#5b3a24" })));
  b.add(new THREE.SphereGeometry(0.9, 12, 10).scale(1.4, 1, 0.8).translate(24.4, 10.8, 0), lam("vasaLion", () => ({ color: "#e0aa3e" })));
  // ---- gun ports, lids, guns ----
  const portM = lam("vasaPort", () => ({ color: "#140c0a" }));
  const lidOut = lam("vasaLidOut", () => ({ color: "#5b3a24" })), lidIn = lam("vasaLidIn", () => ({ color: "#c2322a" }));
  const gunM = lam("vasaGun", () => ({ color: "#3a3532" }));
  const lidGeoO = new THREE.PlaneGeometry(0.9, 0.9).translate(0, -0.45, 0.03), lidGeoI = new THREE.PlaneGeometry(0.86, 0.86).translate(0, -0.45, 0.0).rotateY(Math.PI);
  const gunGeo = new THREE.CylinderGeometry(0.17, 0.22, 1.6, 10).rotateX(Math.PI / 2);
  const lids: Ship["lids"] = [], guns: THREE.Object3D[] = [], ports: THREE.Vector3[] = [];
  PORT_ROWS.forEach((y, row) => PORT_XS.forEach((x0, k) => {
    const x = x0 + (row ? 1.2 : 0);
    for (const s of [-1, 1]) {
      const z = s * hullHalf(x, y), tilt = Math.atan2(hullHalf(x, y + 0.5) - hullHalf(x, y - 0.5), 1);
      b.add(new THREE.PlaneGeometry(0.82, 0.82).translate(0, 0, 0.04).rotateX(-tilt).rotateY(s > 0 ? 0 : Math.PI).translate(x, y, z), portM);
      ports.push(V(x, y, z));
      const hinge = new THREE.Object3D(); hinge.position.set(x, y + 0.45, z + s * 0.05); hinge.rotation.order = "YXZ"; hinge.rotation.y = s > 0 ? 0 : Math.PI;
      const mo = new THREE.Mesh(lidGeoO, lidOut), mi = new THREE.Mesh(lidGeoI, lidIn);
      for (const m of [mo, mi]) { m.castShadow = true; m.receiveShadow = true; hinge.add(m); }
      g.add(hinge); lids.push({ o: hinge, s, k: row * 10 + k });
      const gun = new THREE.Mesh(gunGeo, gunM); gun.position.set(x, y - 0.05, z + s * 0.55); gun.castShadow = true; gun.visible = false; g.add(gun); guns.push(gun);
    }
  }));
  // ---- masts: stumps (always) + upper masts, yards, sails, rigging (rig group) ----
  const mastM = lam("vasaMast", () => ({ color: "#8a5a34" })), ropeM = lam("vasaRope", () => ({ color: "#2a2018" }));
  const sailM = lam("vasaSailM", () => ({ map: sailTex(), side: THREE.DoubleSide }));
  const sails: THREE.Mesh[] = [], furls: THREE.Object3D[] = [];
  const MASTS: [number, number, number, number[]][] = [[11, FORE_Y - 1.8, 24, [11, 19]], [1, DECK_Y, 30, [13, 22]], [-15, POOP_Y, 16, [9]]];
  // sail: a plane hanging below a yard, bulging toward +x by the mesh's scale.x
  const sailGeo = (w: number, h: number) => {
    const geo = new THREE.PlaneGeometry(w, h, 10, 8).rotateY(Math.PI / 2);
    const p = geo.attributes.position;
    for (let k = 0; k < p.count; k++) { const z = p.getZ(k), u = (h / 2 - p.getY(k)) / h; p.setX(k, Math.cos((z / w) * Math.PI) * (Math.sin(Math.PI * u) * 0.85 + 0.15 * u)); }
    geo.computeVertexNormals();
    return geo.translate(0, -h / 2, 0);
  };
  for (const [x, y0, H, yards] of MASTS) {
    b.add(new THREE.CylinderGeometry(0.42, 0.5, 6.5, 12).translate(x, y0 + 3.2, 0), mastM);
    rb.add(new THREE.CylinderGeometry(0.2, 0.42, H - 6, 10).translate(x, y0 + 6.5 + (H - 6) / 2 - 0.5, 0), mastM);
    rb.add(new THREE.CylinderGeometry(1.4, 1.2, 0.35, 14).translate(x, y0 + H * 0.55, 0), mastM); // top platform
    yards.forEach((yy, i) => {
      const wy = (H > 20 ? 16 : 10) * (i ? 0.7 : 1), yh = y0 + yy;
      rb.add(new THREE.CylinderGeometry(0.16, 0.16, wy, 8).rotateX(Math.PI / 2).translate(x + 0.4, yh, 0), mastM);
      const hh = i === 0 ? yy - 2.4 : yards[i] - yards[i - 1] - 1.2, sail = new THREE.Mesh(sailGeo(wy * 0.94, hh), sailM);
      sail.position.set(x + 0.5, yh - 0.1, 0); sail.castShadow = true; sail.receiveShadow = true; rig.add(sail); sails.push(sail);
      const furl = new THREE.Mesh(new THREE.CylinderGeometry(0.42, 0.42, wy * 0.9, 10).rotateX(Math.PI / 2), sailM);
      furl.position.set(x + 0.5, yh - 0.4, 0); furl.castShadow = true; rig.add(furl); furls.push(furl);
    });
    // shrouds to the rails on both sides
    for (const s of [-1, 1]) for (let k = -1; k <= 1; k++) {
      const top = V(x, y0 + H * 0.55, s * 0.6), bx = x + k * 1.2, ty = hullTop(bx);
      rope(rb, ropeM, top, V(bx, ty, s * hullHalf(bx, ty)), 0.05);
    }
    // pennant at the top
    const pen = new THREE.PlaneGeometry(5, 0.5).translate(2.5, 0, 0);
    const pp = pen.attributes.position; for (let k = 0; k < pp.count; k++) { const u = pp.getX(k) / 5; pp.setY(k, pp.getY(k) * (1 - u)); pp.setZ(k, 0.4 * Math.sin(u * 5)); }
    fb.add(pen.translate(x, y0 + H + 0.3, 0), lam("vasaPennant", () => ({ color: "#f3c63a", side: THREE.DoubleSide })));
  }
  // stays + bowsprit
  rope(rb, ropeM, V(1, DECK_Y + 30 * 0.55, 0), V(11, FORE_Y - 1.8 + 6, 0), 0.07);
  rope(rb, ropeM, V(11, FORE_Y - 1.8 + 24 * 0.55, 0), V(28, 14, 0), 0.07);
  rb.add(new THREE.CylinderGeometry(0.25, 0.4, 13, 10).rotateZ(-1.15).translate(23, 11.3, 0), mastM);
  // ensign at the stern
  fb.add(new THREE.CylinderGeometry(0.1, 0.12, 8, 6).translate(-20.6, POOP_Y + 4, 0), mastM);
  {
    const fl = new THREE.PlaneGeometry(6, 3.8, 12, 1).translate(-3, 0, 0);
    const p = fl.attributes.position; for (let k = 0; k < p.count; k++) p.setZ(k, 0.45 * Math.sin(p.getX(k) * 1.2));
    fb.add(fl.translate(-20.7, POOP_Y + 6, 0), lam("flagSE", () => ({ map: flagTexSE(), side: THREE.DoubleSide })));
  }
  // salvage cables (seabed raise)
  for (const [x, s] of [[-10, 1], [-10, -1], [9, 1], [9, -1]] as const) rope(cb, lam("vasaCable", () => ({ color: "#2a2a30" })), V(x, -3.5, s * hullHalf(x, -3.5)), V(x, 70, s * 7), 0.32);
  b.build(g); rb.build(rig); fb.build(flags); cb.build(cables);
  return { g, rig, flags, cables, lids, guns, sails, furls, ports };
};
/** pose the ship for a frame: transform, open lids (staggered), guns, sails set / furled + billow, flags, cables */
export const poseShip = (sh: Ship, p: STORY.ShipPose) => {
  sh.g.position.set(p.x, p.y, p.z); sh.g.rotation.set(p.roll, 0, p.pitch);
  for (const l of sh.lids) {
    const k = STORY.clamp01(p.ports * 1.6 - (l.k % 10) * 0.06 - (l.k >= 10 ? 0.1 : 0));
    l.o.rotation.x = -k * 1.75;
  }
  sh.guns.forEach((gn, i) => { gn.visible = p.ports > 0.3 && STORY.clamp01(sh.lids[i].o.rotation.x / -1.2) > 0.5; });
  sh.rig.visible = p.masts; sh.flags.visible = p.flags && p.masts; sh.cables.visible = p.cables;
  for (const s of sh.sails) { s.visible = p.sails > 0.02 && p.masts; s.scale.set(Math.max(0.05, p.billow), Math.max(0.05, p.sails), 1); }
