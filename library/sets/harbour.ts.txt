// LIBRARY PART: harbour dressing for the town: fishing boats on the river, triangle bunting along both quays, striped bakery awning, quay bench, crates and lobster pots
// Append to the END of the short's build.ts (below "library parts appended below"), then register in sets.ts:
//   inside SETS.town after townSet: buildHarbour(c.root, c.rays, c.glow, { boats: BOATS, baker: BAKER, bench: BENCH });
// Needs: the town set; story.ts D0 + BOATS [x, z, yaw, colour][], BAKER, BENCH positions (any [x,y,z]).
// Origin: Two Suns
// ======================= Two Suns: harbour, sea, bedroom, Venus =======================
const BOAT_COLS = ["#c0453a", "#2f5fa8", "#2f8f6a", "#f2c230"];
const boatHull = (col: string) => lam("hull" + col, () => ({ color: col, side: THREE.DoubleSide }));
/** small fishing boat: painted hull, white cabin, mast; origin at the waterline */
export const fishingBoat = (b: Batch, x: number, z: number, yaw: number, ci: number, y = 0) => {
  const m = new THREE.Matrix4().makeRotationY(yaw).setPosition(x, y, z);
  const hull = canoeGeo(6.4, 2.3, 1.0);
  b.add(hull.clone().translate(0, 0.62, 0), boatHull(BOAT_COLS[ci % 4]), m);
  b.add(boxGeo(V(-1.1, 0.5, -3.1), V(1.1, 0.66, 3.1)), lam("deckWood", () => ({ color: "#c9965e" })), m);
  b.add(boxGeo(V(-0.75, 0.66, -0.6), V(0.75, 2.0, 1.2)), lam("cabinWhite", () => ({ color: "#f4f1ea" })), m);
  b.add(boxGeo(V(-0.85, 2.0, -0.7), V(0.85, 2.12, 1.3)), lam("cabinRoof", () => ({ color: "#2f4f8a" })), m);
  b.add(new THREE.CylinderGeometry(0.06, 0.08, 4.2, 6).translate(0, 2.7, -1.6), lam("mast", () => ({ color: "#8a5a3a" })), m);
  b.add(boxGeo(V(-0.6, 0.66, -2.8), V(0.6, 1.0, -1.9)), lam("crate", () => ({ color: "#a8724a" })), m);
};
const buntTex = (col: string) => ctex("bunt" + col, 64, 80, (g) => {
  g.clearRect(0, 0, 64, 80);
  g.fillStyle = col; g.beginPath(); g.moveTo(0, 0); g.lineTo(64, 0); g.lineTo(32, 78); g.closePath(); g.fill();
  g.strokeStyle = "rgba(36,23,38,0.6)"; g.lineWidth = 3; g.stroke();
});
const BUNT = ["#e2453c", "#f2c230", "#3f7fd8", "#f4f1ea", "#2fa36b"];
const buntMat = (col: string) => lam("bunt" + col, () => ({ map: buntTex(col), alphaTest: 0.5, side: THREE.DoubleSide }));
/** a sagging string of triangle bunting between a and c (fairy bulbs to the glow batch) */
const bunting = (b: Batch, gb: Batch | null, a: THREE.Vector3, c: THREE.Vector3, sag = 0.8) => {
  const n = Math.max(4, Math.round(a.distanceTo(c) / 0.6)), cord = lam("cord", () => ({ color: "#2a2430" }));
  const pt = (u: number) => a.clone().lerp(c, u).add(V(0, -sag * 4 * u * (1 - u), 0));
  const dir = c.clone().sub(a).setY(0).normalize(), yaw = Math.atan2(dir.x, dir.z) + Math.PI / 2;
  for (let i = 0; i < n; i++) {
    const p0 = pt(i / n), p1 = pt((i + 1) / n);
    b.add(new THREE.CylinderGeometry(0.012, 0.012, p0.distanceTo(p1), 4).applyMatrix4(new THREE.Matrix4().compose(p0.clone().add(p1).multiplyScalar(0.5),
      new THREE.Quaternion().setFromUnitVectors(V(0, 1, 0), p1.clone().sub(p0).normalize()), V(1, 1, 1))), cord);
    const m = pt((i + 0.5) / n);
    b.add(new THREE.PlaneGeometry(0.34, 0.42).translate(0, -0.22, 0), buntMat(BUNT[i % BUNT.length]), new THREE.Matrix4().compose(m, new THREE.Quaternion().setFromAxisAngle(V(0, 1, 0), yaw), V(1, 1, 1)));
    if (gb && i % 2 === 0) gb.add(new THREE.SphereGeometry(0.06, 8, 6).translate(p0.x, p0.y - 0.06, p0.z), GLOW_WIN);
  }
};
const awnTex = () => ctex("awning", 256, 128, (g) => {
  for (let i = 0; i < 8; i++) { g.fillStyle = i % 2 ? "#f4f1ea" : "#e2453c"; g.fillRect(i * 32, 0, 32, 128); }
  g.fillStyle = "rgba(36,23,38,0.35)"; g.fillRect(0, 118, 256, 10);
});
const crateM = () => lam("crateHD", () => ({ color: "#a8724a" }));
/** lobster pot: half cylinder cage */
const lobsterPot = (b: Batch, x: number, y: number, z: number, s: number) => {
  if (inTheWay(x - 0.5, x + 0.5, y, y + 0.6, z - 0.5, z + 0.5, true)) return;
  b.add(new THREE.CylinderGeometry(0.38 * s, 0.38 * s, 0.9 * s, 10, 1, false, 0, Math.PI).rotateZ(Math.PI / 2).rotateY(Math.PI / 2).translate(x, y, z), lam("potNet", () => ({ color: "#3f7f6a", side: THREE.DoubleSide })));
  b.add(boxGeo(V(x - 0.45 * s, y, z - 0.38 * s), V(x + 0.45 * s, y + 0.06, z + 0.38 * s)), crateM());
};
/** the harbour dressing: boats, bunting along both quays, the bakery awning, the bench, crates and pots */
export const buildHarbour = (root: THREE.Object3D, rays: Ray[], glow: THREE.Object3D, o: { boats: [number, number, number, number][]; baker: [number, number, number]; bench: [number, number, number] }) => {
  const b = new Batch(), gb = new Batch(), r = rng(77);
  sight = rays;
  for (const [x, z, yaw, ci] of o.boats) fishingBoat(b, x, z, Math.PI / 2 + yaw, ci);
  for (const bank of [-1, 1]) {
    for (let x = -126; x < 126; x += 14) if (Math.abs(x) > 6 && Math.abs(x + 14) > 6) bunting(b, gb, V(x, STORY.D0 + 3.3, bank * 31.0), V(x + 14, STORY.D0 + 3.3, bank * 31.0), 0.6);
    for (let z = 42; z < 120; z += 9) bunting(b, gb, V(-6.2, STORY.D0 + 5.2, bank * z), V(6.2, STORY.D0 + 5.2, bank * (z + 2)), 0.8);
    for (let x = -120; x < 120; x += 6 + r() * 8) {
      if (Math.abs(x) < 8 || (bank < 0 && Math.abs(x) < 60)) continue; // keep the near quay clear where the cast stands
      const k = r(), z = bank * (35.4 + r() * 0.6);
      if (k < 0.3) lobsterPot(b, x, STORY.D0, z, 0.9 + r() * 0.3);
      else if (k < 0.5) { const s = 0.5 + r() * 0.2; if (!inTheWay(x - s, x + s, STORY.D0, STORY.D0 + s, z - s, z + s, true)) b.add(boxGeo(V(x - s, STORY.D0, z - s), V(x + s, STORY.D0 + s, z + s)), crateM()); }
    }
  }
  // bakery: striped awning over the door, a bread shelf in the window
  const [bx, , bz] = o.baker;
  const awn = new THREE.PlaneGeometry(3.4, 1.5);
  awn.rotateX(-Math.PI / 2 + 0.5).translate(bx, STORY.D0 + 2.75, bz - 0.55);
  b.add(awn, lam("awningHD", () => ({ map: awnTex(), side: THREE.DoubleSide })));
  for (const dx of [-1.6, 1.6]) b.add(boxGeo(V(bx + dx - 0.04, STORY.D0 + 2.2, bz - 1.3), V(bx + dx + 0.04, STORY.D0 + 3.3, bz - 1.25)), lam("lampIron", () => ({ color: "#2e3240" })));
  // bench on the quay
  const [cx, , cz] = o.bench, wood = lam("benchWood", () => ({ color: "#a8724a" })), iron = lam("lampIron", () => ({ color: "#2e3240" }));
  b.add(boxGeo(V(cx - 1.6, STORY.D0 + 0.42, cz - 0.25), V(cx + 1.6, STORY.D0 + 0.5, cz + 0.25)), wood);
  b.add(boxGeo(V(cx - 1.6, STORY.D0 + 0.55, cz - 0.32), V(cx + 1.6, STORY.D0 + 0.95, cz - 0.26)), wood);
  for (const dx of [-1.4, 1.4]) b.add(boxGeo(V(cx + dx - 0.05, STORY.D0, cz - 0.3), V(cx + dx + 0.05, STORY.D0 + 0.95, cz + 0.25)), iron);
  b.build(root);
  gb.build(glow, false);
  sight = [];
};
