// ---- geometry helpers ----
const DS = THREE.DoubleSide;
const jm = (key: string, p: () => THREE.MeshLambertMaterialParameters) => lam("j_" + key, () => ({ side: DS, ...p() }));
/** quad a-b-c-d oriented so its normal points along `want` */
const face = (a: THREE.Vector3, b: THREE.Vector3, c: THREE.Vector3, d: THREE.Vector3, want: THREE.Vector3, u = 1, v = 1) => {
  const n = new THREE.Vector3().subVectors(b, a).cross(new THREE.Vector3().subVectors(d, a));
  return n.dot(want) >= 0 ? quad(a, b, c, d, u, v) : quad(b, a, d, c, u, v);
};
/** hipped roof / eave skirt: square frustum from a CylinderGeometry with 4 sides, flat normals, scaled to a rectangle */
const hip = (cx: number, y0: number, cz: number, hx: number, hz: number, h: number, top = 0.3) => {
  const g = new THREE.CylinderGeometry(top, 1, h, 4, 1, true).rotateY(Math.PI / 4).toNonIndexed();
  g.scale(hx * Math.SQRT2, 1, hz * Math.SQRT2); g.translate(cx, y0 + h / 2, cz); g.computeVertexNormals();
  return g;
};
/** hipped roof with a ridge along x: eave rectangle (hx, hz) at y0, ridge of half-length rl at y0 + h; flat faces */
const hipRoof = (b: Batch, m: THREE.Material, cx: number, y0: number, cz: number, hx: number, hz: number, h: number, rl: number) => {
  const y1 = y0 + h, R0 = V(cx - rl, y1, cz), R1 = V(cx + rl, y1, cz);
  const A = V(cx - hx, y0, cz + hz), B = V(cx + hx, y0, cz + hz), C = V(cx + hx, y0, cz - hz), D = V(cx - hx, y0, cz - hz);
  b.add(face(A, B, R1, R0, V(0, 1, 1), hx / 2, 1.5), m);
  b.add(face(C, D, R0, R1, V(0, 1, -1), hx / 2, 1.5), m);
  const t1 = tri(B, C, R1), t2 = tri(D, A, R0);
  b.add(t1, m); b.add(t2, m);
};
const blossomMat = () => jm("blossom", () => ({ color: "#f3b6c8" }));
/** cherry tree in bloom: trunk + pink lobes */
const addBlossom = (b: Batch, x: number, y: number, z: number, h: number, seed: number) => {
  if (inTheWay(x - 0.4 * h, x + 0.4 * h, y, y + h, z - 0.4 * h, z + 0.4 * h, true)) return;
  const r = rng(seed), lobe = lobeGeo(0);
  b.add(trunkGeo(), barkMat(), new THREE.Matrix4().compose(V(x, y, z), new THREE.Quaternion(), V(0.05 * h, 0.55 * h, 0.05 * h)));
  for (let i = 0; i < 7; i++) {
    const a = r() * 6.3, d = i ? 0.22 * h : 0, rad = (0.16 + r() * 0.08) * h;
    b.add(lobe, blossomMat(), new THREE.Matrix4().compose(V(x + Math.cos(a) * d, y + 0.62 * h + (r() - 0.4) * 0.15 * h, z + Math.sin(a) * d), new THREE.Quaternion(), V(rad, rad * 0.8, rad)));

const hill = (b: Batch, x: number, z: number, sx: number, sy: number, sz: number, col: string) =>

/** a Kyoto townhouse (machiya) with its street face on x = xs, facing the street (side +1 = right row, faces -x) */
const machiya = (b: Batch, gb: Batch, xs: number, side: number, z0: number, z1: number, seed: number) => {
  const r = rng(seed), dep = 7, h1 = 3.1, h2 = 5.2, ridge = 6.7 + r() * 0.4, xb = xs + side * dep;
  const lo = Math.min(xs, xb), hi = Math.max(xs, xb);
  if (inTheWay(lo, hi, 0, ridge, z0, z1)) return;
  const want = V(-side, 0, 0), len = z1 - z0, cols = ["#5b3b28", "#7a3a2a", "#4a3a30", "#6a4a32"];
  b.add(boxGeo(V(lo, 0, z0), V(hi, h2, z1)), jm("plaster", () => ({ map: rep(jPlaster(), 2, 2) })));
  const fx = xs - side * 0.03;
  b.add(face(V(fx, 0, z0), V(fx, 0, z1), V(fx, h1, z1), V(fx, h1, z0), want), jm("lattice" + (seed % 4), () => ({ map: jLattice(cols[seed % 4]) })));
  b.add(face(V(fx, h1, z0), V(fx, h1, z1), V(fx, h2, z1), V(fx, h2, z0), want), jm("upper", () => ({ map: jUpper() })));
  const roofM = jm("roofTile", () => ({ map: rep(roofHD("#4c5262"), 3, 3) }));
  // lean-to eave over the ground floor
  const e = xs - side * 0.95;
  b.add(face(V(xs, 3.55, z0 - 0.1), V(xs, 3.55, z1 + 0.1), V(e, 3.05, z1 + 0.1), V(e, 3.05, z0 - 0.1), V(-side * 0.4, 1, 0), len / 3, 1), roofM);
  // main gable roof, ridge along z
  const xr = xs + side * dep / 2, ef = xs - side * 0.5, eb = xb + side * 0.5;
  b.add(face(V(ef, h2 - 0.1, z0 - 0.15), V(ef, h2 - 0.1, z1 + 0.15), V(xr, ridge, z1 + 0.15), V(xr, ridge, z0 - 0.15), V(-side * 0.5, 1, 0), len / 3, 1.5), roofM);
  b.add(face(V(eb, h2 - 0.1, z0 - 0.15), V(eb, h2 - 0.1, z1 + 0.15), V(xr, ridge, z1 + 0.15), V(xr, ridge, z0 - 0.15), V(side * 0.5, 1, 0), len / 3, 1.5), roofM);
  for (const zz of [z0 - 0.01, z1 + 0.01]) b.add(tri(V(xs, h2, zz), V(xb, h2, zz), V(xr, ridge - 0.05, zz)), jm("plaster", () => ({ map: rep(jPlaster(), 2, 2) })));
  b.add(boxGeo(V(xr - 0.18, ridge - 0.1, z0 - 0.15), V(xr + 0.18, ridge + 0.22, z1 + 0.15)), jm("ridge", () => ({ color: "#3a3e4a" })));
  // noren at the door, a red paper lantern under the eave
  const zc = z0 + len * 0.47, nx = xs - side * 0.06;
  if (r() < 0.7) b.add(face(V(nx, 1.7, zc - 0.8), V(nx, 1.7, zc + 0.8), V(nx, 2.75, zc + 0.8), V(nx, 2.75, zc - 0.8), want), jm("noren", () => ({ map: jNoren() })));
  if (r() < 0.6) {
    const lx = xs - side * 0.55, lz = z0 + len * (r() < 0.5 ? 0.18 : 0.82);
    b.add(new THREE.CylinderGeometry(0.2, 0.2, 0.5, 12).translate(lx, 2.55, lz), jm("lantern", () => ({ color: "#d0402e" })));
    gb.add(new THREE.CylinderGeometry(0.205, 0.205, 0.36, 12).translate(lx, 2.55, lz), GLOW_FIRE);
  }
};
