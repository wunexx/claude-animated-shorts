/** a big cast-iron clock weight (held overhead) on a 400 x 500 canvas */

  p.ell(200, 70, 34, 30, null, 9); p.ell(200, 70, 34, 30, null, 5);
  p.g.strokeStyle = "#c9962e"; p.g.lineWidth = 12; p.g.beginPath(); p.g.ellipse(200, 70, 30, 26, 0, 0, 7); p.g.stroke();

  p.poly([[78, 150], [130, 150], [122, 420], [70, 420]], "#5c6272", 0);

/** the lens's face (hallucination): k 0 closed .. 1 eyes open + a grin, on a 420 x 260 canvas (transparent) */
export const fxLensFace = (p: Pen, k: number) => {
  for (const x of [120, 300]) {
    p.ell(x, 100, 62, 10 + 52 * k, "#ffffff", 9);
    if (k > 0.3) { p.dot(x + 14, 110, 24 * k, INK); p.dot(x + 22, 98, 7 * k, "#ffffff"); }
    p.stroke([[x - 60, 30 - 20 * k], [x + 50, 22 - 26 * k]], 11, INK);
  }
  if (k > 0.2) { p.poly([[90, 190], [330, 190], [300, 232], [120, 232]], "#7a1e2a", 9); p.poly([[110, 190], [310, 190], [302, 204], [118, 204]], "#ffffff", 0); }
};
