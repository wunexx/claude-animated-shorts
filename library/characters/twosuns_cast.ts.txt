/** our dog, side view on the 600 x 1000 design card, facing right (mirror for left); paws at y 975 */
export type DogOpts = { phase?: number; face?: "alert" | "happy" | "dizzy" | "pant" | "worried"; mode?: "run" | "stand" | "flat" | "sit"; wag?: number; blink?: number };
const DOG = "#d99a5a", DOG_SH = "#b07440", DOG_PATCH = "#7a4a2a";
export const dog = (p: Pen, o: DogOpts) => {
  const { phase = 0, face = "alert", mode = "stand", wag = 0, blink = 0 } = o, two = Math.PI * 2;
  const flat = mode === "flat", sit = mode === "sit";
  const by = flat ? 922 : sit ? 860 : 848 + (mode === "run" ? 8 * Math.cos(two * phase * 2) : 0);
  const ow = 7;
  const tw = Math.sin(two * wag) * 26;
  p.line(curve([200, by - 20], [140, by - 70 + tw * 0.4], [118 + tw, by - 130]), DOG, 20);
  const legs = (far: boolean) => {
    const col = far ? DOG_SH : DOG;
    for (const [x, ph] of (far ? [[262, 0.25], [392, 0.75]] : [[236, 0.5], [366, 0]]) as [number, number][]) {
      if (flat) { p.line([[x, by + 20], [x + (x > 300 ? 60 : -60), 962]], col, 22); continue; }
      if (sit && x < 300) { p.line([[x + 10, by + 30], [x + 60, 960]], col, 24); continue; }
      const sw = mode === "run" ? Math.sin(two * (phase + ph)) : 0;
      p.line([[x, by + 30], [x + 34 * sw, 968 - Math.max(0, sw) * 20]], col, 22);
    }
  };
  legs(true);
  const rot = sit ? -18 : 0, ry = flat ? 50 : 62;
  p.ell(300, by, 135, ry, DOG, ow, 0, 360, rot);
  p.clip(ellPts(300, by, 135, ry, 0, 360, rot), () => { p.ell(250, by - 30, 50, 32, DOG_PATCH, 0); p.ell(300, by + 70, 160, 40, DOG_SH, 0); });
  p.ell(300, by, 135, ry, null, ow, 0, 360, rot);
  legs(false);
  const hx = flat ? 455 : 448, hy = flat ? by - 30 : by - 92;
  p.ell(hx + 62, hy + 22, 48, 32, "#f0c08a", ow);
  p.ell(hx, hy, 70, 64, DOG, ow);
  p.ell(hx - 34, hy + 26, 26, 52, DOG_PATCH, ow, 0, 360, 18);
  p.dot(hx + 104, hy + 10, 12);
  const ex = hx + 30, ey = hy - 14;
  if (face === "dizzy") { for (let k = 0; k < 2; k++) p.arc(ex - 6 + k * 3, ey, 14 - k * 7, 14 - k * 7, 0, 300, 4); }
  else if (face === "happy") p.arc(ex, ey + 6, 13, 10, 200, 340, 6);
  else if (face === "pant") { p.ell(ex, ey, 9, 7, INK, 0, 0, 180); p.stroke([[ex - 13, ey], [ex + 13, ey]], 6); }
  else if (blink > 0.6) p.stroke([[ex - 11, ey], [ex + 11, ey]], 6);
  else p.ell(ex, ey, face === "worried" ? 7 : 9, face === "worried" ? 9 : 11, INK, 0);
  if (face === "worried") p.stroke([[ex - 14, ey - 26], [ex + 12, ey - 20]], 5);
  if (face === "happy" || face === "pant" || face === "dizzy") {
    p.poly([[hx + 50, hy + 44], [hx + 92, hy + 40], [hx + 70, hy + 62]], MOUTH, 5);
    p.ell(hx + 70, hy + 70, 13, face === "pant" ? 24 : 17, TONGUE, 5);
  } else p.arc(hx + 72, hy + 38, 18, 9, 30, 150, 5);
};

/** rising steam wisps on a 256 x 512 canvas, see-through; a = strength */
export const fxSteam = (p: Pen, seed: number, a = 0.6) => {
  const g = p.g;
  g.save(); g.filter = "blur(10px)"; g.lineCap = "round";
  for (let k = 0; k < 3; k++) {
    const x0 = 70 + k * 58 + ((seed * 31 + k * 17) % 20);
    g.strokeStyle = "rgba(255,255,255," + (a * (0.9 - 0.2 * k)).toFixed(3) + ")"; g.lineWidth = 34 - 6 * k;
    g.beginPath();
    for (let i = 0; i <= 24; i++) { const y = 500 - i * 20, x = x0 + 26 * Math.sin(i * 0.55 + seed + k * 2); if (i) g.lineTo(x, y); else g.moveTo(x, y); }
    g.stroke();
  }
  g.restore();
  const fade = g.createLinearGradient(0, 0, 0, 512);
  fade.addColorStop(0, "rgba(0,0,0,1)"); fade.addColorStop(0.45, "rgba(0,0,0,0)");
  g.globalCompositeOperation = "destination-out"; g.fillStyle = fade; g.fillRect(0, 0, 256, 512); g.globalCompositeOperation = "source-over";
};
/** sunbeam strip on a 128 x 1024 canvas (bright top, fading toward the bottom and the edges), see-through */
export const fxBeam = (p: Pen) => {
  const g = p.g;
  for (let y = 0; y < 1024; y += 4) {
    const v = y / 1024, a = 0.55 * Math.sin(Math.PI * Math.min(1, v * 1.4 + 0.05)) * (1 - 0.5 * v);
    const gr = g.createLinearGradient(0, 0, 128, 0);
    gr.addColorStop(0, "rgba(255,236,170,0)"); gr.addColorStop(0.5, "rgba(255,240,190," + a.toFixed(3) + ")"); gr.addColorStop(1, "rgba(255,236,170,0)");
    g.fillStyle = gr; g.fillRect(0, y, 128, 4);
  }
};

/** the sleeper in bed, side-on, 1000 x 700 canvas: n blankets piled on, face */
const QUILTS = ["#e2457a", "#3f7fd8", "#f2c230", "#2fa36b", "#8a4fc9", "#f08a2c"];
export const fxBed = (p: Pen, n: number, expr: Expr, lid = 0, light = DEF_LIGHT) => {
  p.poly(rrect(40, 300, 110, 690, 18), "#a8724a", 10);
  p.poly(rrect(880, 430, 940, 690, 16), "#a8724a", 10);
  p.poly(rrect(70, 540, 920, 600, 12), "#8a5a3a", 10);
  p.poly(rrect(90, 470, 900, 545, 22), "#f4f1ea", 10);
  p.poly(rrect(110, 400, 300, 478, 36), "#ffffff", 10);
  const H = headBlob(4, 74, 1), hc = 380;
  const hp = blobPts(210, hc, H);
  p.shadedFill(hp, SKIN, mul(SKIN, SH_SKIN), light, H.a);
  p.inkRing(hp, 9, thickAngle(light));
  hatOn(p, { jacket: "#fff", pants: "#fff", boots: "#fff", hat: "hair", hair: "#5a3b2a" }, hp, H, 210, hc, light, 4, 1, false);
  faceFront(p, 210, hc, 2 * H.a, expr, 1, 0, lid, false, SKIN);
  for (let i = 0; i < n; i++) { // a puffy pile: each quilt fat and bulging, a little narrower than the one under it
    const y0 = 480 - i * 50, col = QUILTS[i % QUILTS.length], x0 = 262 + i * 14, x1 = 892 - i * 10;
    const pts: P2[] = [[x0, y0 + 66], [x0 + 6, y0 + 14], [x0 + 120, y0 - 26], [(x0 + x1) / 2, y0 - 34], [x1 - 110, y0 - 24], [x1 - 6, y0 + 10], [x1, y0 + 68]];
    p.poly(curveClose(pts), col, 10);
    for (let k = 1; k < 5; k++) { const x = x0 + ((x1 - x0) * k) / 5; p.stroke([[x, y0 - 22], [x - 6, y0 + 56]], 3, "rgba(36,23,38,0.4)"); }
    p.stroke([[x0 + 40, y0 + 2], [x1 - 60, y0 - 8]], 6, "rgba(255,255,255,0.35)");
  }
};

/** a lead fishing sinker on hot ground, 400 x 400 canvas; m = 0 solid .. 1 melted to a shiny puddle */
export const fxSinker = (p: Pen, m: number) => {
  const yb = 360, pr = 34 + 150 * m, ph = 10 + 26 * m, slump = Math.min(1, m * 1.15);
  // fishing line from the eyelet up out of frame (goes slack as it melts)
  const h = 250 * (1 - 0.85 * slump), w = 46 + 34 * slump, top = yb - h;
  p.stroke(curve([200, top - 34], [200 + 40 * slump, (top - 34) / 2], [200 + 10 * slump, 0]), 3, "rgba(240,240,250,0.85)");
  p.ell(200, yb, pr, ph, mix("#9aa2aa", "#ffe2b0", 0.45 * m), 7);                 // shiny molten puddle
  p.ell(200 - pr * 0.35, yb - ph * 0.3, pr * 0.28, ph * 0.3, "rgba(255,255,255,0.65)", 0);
  if (m < 0.98) {                                                                   // the teardrop sinker, slumping
    const body: P2[] = [[200, top], [200 + w * 0.32, top + h * 0.25], [200 + w, yb - h * 0.22], [200 + w * 0.9, yb - 4], [200 - w * 0.9, yb - 4], [200 - w, yb - h * 0.22], [200 - w * 0.32, top + h * 0.25]];
    p.poly(curveClose(body), mix("#7d848c", "#d9a070", 0.5 * m), 9);
    p.ell(200 - w * 0.4, top + h * 0.55, 9, h * 0.2, "rgba(255,255,255,0.4)", 0);
    if (m < 0.85) p.arc(200, top - 18, 14, 16, 0, 360, 8, "#5f666e");          // eyelet
  }
};

/** dark towering storm cloud on a 1024 x 640 canvas (soft card) */
export const fxCloudBank = (p: Pen, seed: number) => {
  const g = p.g;
  const puffs: [number, number, number][] = [];
  for (let i = 0; i < 16; i++) { const u = i / 15; puffs.push([90 + 840 * u + ((seed * 53 + i * 37) % 60) - 30, 520 - 380 * Math.sin(Math.PI * u) * (0.6 + 0.4 * ((i * 7 + seed) % 5) / 5), 90 + ((i * 29 + seed * 11) % 70)]); }
  for (const [x, y, r] of puffs) {
    const gr = g.createRadialGradient(x - r * 0.3, y - r * 0.35, r * 0.1, x, y, r);
    gr.addColorStop(0, "rgba(120,124,150,1)"); gr.addColorStop(0.75, "rgba(70,72,96,1)"); gr.addColorStop(1, "rgba(52,52,74,0)");
    g.fillStyle = gr; g.beginPath(); g.arc(x, y, r, 0, Math.PI * 2); g.fill();
  }
  const base = g.createLinearGradient(0, 400, 0, 640);
  base.addColorStop(0, "rgba(58,58,82,1)"); base.addColorStop(1, "rgba(58,58,82,0)");
  g.fillStyle = base; g.beginPath(); g.ellipse(512, 520, 470, 110, 0, 0, Math.PI * 2); g.fill();
};
