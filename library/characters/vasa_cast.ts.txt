/** a sealed letter (parchment, rolled ends, writing, red wax seal); hurry = big red HURRY! across it */
export const fxLetter = (p: Pen, hurry: boolean) => {
  const g = p.g, paper = "#f2e3bf", edge = "#c9a96a";
  p.poly([[90, 90], [510, 90], [520, 720], [80, 720]], paper, 7);
  p.ell(300, 90, 225, 30, mix(paper, edge, 0.5), 6); p.ell(300, 720, 225, 30, mix(paper, edge, 0.5), 6);
  for (let i = 0; i < 9; i++) {
    const y = 170 + i * 44, w = i === 8 ? 180 : 340 - (i % 3) * 30;
    const pts: [number, number][] = [];
    for (let x = 0; x <= w; x += 12) pts.push([130 + x, y + Math.sin(x * 0.21 + i) * 5]);
    p.stroke(pts, 4.5, "#4a3a5a");
  }
  p.ell(420, 610, 52, 48, "#b8282a", 5); p.ell(420, 610, 30, 28, "#d23a36", 0);
  if (hurry) {
    g.save(); g.translate(290, 470); g.rotate(-0.16);
    g.font = "bold 112px Georgia, serif"; g.textAlign = "center"; g.textBaseline = "middle";
    g.lineWidth = 16; g.strokeStyle = INK; g.strokeText("HURRY!", 0, 0);
    g.fillStyle = "#e0322a"; g.fillText("HURRY!", 0, 0);
    g.restore();
  }
};

/** cross-section of the ship: two gun decks high up, a little ballast low down, the waterline level; o.guns/ballast/cg
 *  highlight 0..1, o.tilt (rad) rolls the hull about the waterline */
export const fxSection = (p: Pen, o: { guns: number; ballast: number; cg: number; tilt: number }) => {
  const g = p.g, cx = 500, wl = 600;
  g.fillStyle = "rgba(110,190,255,0.35)"; g.fillRect(0, wl, 1000, 400);
  g.translate(500, 500); g.scale(0.8, 0.8); g.translate(-500, -500);
  g.save(); g.translate(cx, wl); g.rotate(o.tilt); g.translate(-cx, -wl);
  const hw = (y: number) => { const v = (y - 820) / -560; return 70 + 260 * Math.pow(Math.sin(Math.min(1, Math.max(0, v * 1.6)) * Math.PI / 2), 0.7) - (v > 0.6 ? (v - 0.6) * 140 : 0); };
  const L: [number, number][] = [], R: [number, number][] = [];
  for (let y = 820; y >= 230; y -= 15) { L.push([cx - hw(y), y]); R.unshift([cx + hw(y), y]); }
  p.poly([...L, ...R], "#a8744a", 9);
  p.poly([...L.map(([x, y]) => [x + 26, y] as [number, number]).slice(1), ...R.map(([x, y]) => [x - 26, y] as [number, number]).slice(0, -1)], "#6b4528", 0);
  for (const y of [320, 450]) { p.stroke([[cx - hw(y) + 20, y + 40], [cx + hw(y) - 20, y + 40]], 10, "#3a2416"); }
  p.stroke([[cx - hw(600) + 20, 640], [cx + hw(600) - 20, 640]], 8, "#3a2416");
  const glow = (k: number, col: string) => (k > 0 ? mix(col, "#ffe066", 0.5 * k) : col);
  for (const y of [400, 270]) for (const s of [-1, 1]) {
    const x = cx + s * (hw(y + 40) - 75);
    p.ell(x, y + 10, 52, 30, glow(o.guns, "#2e2a2e"), 6); p.ell(x - s * 30, y + 26, 22, 22, "#6b4528", 5);
    p.poly([[x + s * 40, y - 4], [x + s * 112, y - 6], [x + s * 112, y + 22], [x + s * 40, y + 24]], glow(o.guns, "#2e2a2e"), 6);
  }
  for (let i = 0; i < 5; i++) p.ell(cx - 50 + i * 25, 770 - (i % 2) * 10, 20, 15, glow(o.ballast, "#8a8a90"), 5);
  p.stroke([[cx, 230], [cx, 70]], 14, "#8a5a34");
  if (o.cg > 0) {
    const y = 330, r = 34 * Math.min(1, o.cg * 1.5);
    p.ell(cx, y, r, r, "#e0322a", 7);
    if (o.cg > 0.5) { p.stroke([[cx, y + r + 8], [cx, y + r + 90]], 12, "#e0322a"); p.poly([[cx - 28, y + r + 80], [cx + 28, y + r + 80], [cx, y + r + 120]], "#e0322a", 5); }
  }
  g.restore();
  p.stroke([[-200, wl], [1200, wl]], 8, "#dff2ff");
};

/** a big rolling gun-smoke cloud, k 0 (fresh, dense) .. 1 (spread, faded) */
export const fxSmoke = (p: Pen, k: number) => {
  const g = p.g, a = 0.85 * (1 - k) ** 1.2;
  for (let i = 0; i < 11; i++) {
    const ang = i * 2.4, d = 60 + 110 * Math.sqrt(k) * ((i % 3) / 2 + 0.4), r = 70 + 40 * ((i * 7) % 5) / 5 + 50 * k;
    g.fillStyle = `rgba(${236 - i * 3},${232 - i * 3},${224 - i * 2},${a.toFixed(3)})`;
    g.beginPath(); g.arc(256 + Math.cos(ang) * d, 256 + Math.sin(ang) * d * 0.7, r, 0, 7); g.fill();
  }
};
