/** a wreath of leaves with a red ribbon, 256 x 256 */
export const fxWreath = (p: Pen, seed = 0) => {
  for (let i = 0; i < 16; i++) {
    const a = (i / 16) * Math.PI * 2 + seed, x = 128 + Math.cos(a) * 82, y = 128 + Math.sin(a) * 82;
    p.ell(x, y, 26, 13, i % 2 ? "#5fa34a" : "#3f8a3a", 4, 0, 360, (a * 180) / Math.PI + 90);
  }
  for (let i = 0; i < 6; i++) { const a = (i / 6) * Math.PI * 2 + 0.3 + seed; p.dot(128 + Math.cos(a) * 82, 128 + Math.sin(a) * 82, 9, ["#e0453a", "#f2d24a", "#f4f1ea"][i % 3]); }
  p.poly([[108, 200], [128, 214], [148, 200], [160, 252], [140, 240], [128, 256], [116, 240], [96, 252]], "#c0353a", 4);
};
