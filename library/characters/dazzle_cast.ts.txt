const dzBands = (g: CanvasRenderingContext2D, x0: number, y0: number, w: number, h: number, seed: number) => {
  let s = seed; const r = () => ((s = (s * 16807) % 2147483647) - 1) / 2147483646;
  const cols = ["#15171c", "#25375f", "#6f8fb8", "#15171c"];
  let x = x0 - h, i = 0;
  while (x < x0 + w + h) {
    const bw = h * (0.25 + r() * 0.35), sk = h * (r() < 0.5 ? 0.6 : -0.5);
    g.fillStyle = cols[i++ % 4]; g.beginPath(); g.moveTo(x, y0); g.lineTo(x + bw, y0); g.lineTo(x + bw + sk, y0 + h); g.lineTo(x + sk, y0 + h); g.closePath(); g.fill();
    x += bw * (1.6 + r() * 0.8);
  }
};
/** Wilkinson's sketch [600x420]: a ship side view in clashing stripes on cream paper with pencil notes */
export const fxSketch = (p: Pen) => {
  const g = p.g;
  p.poly(rrect(10, 10, 590, 410, 18), "#f4ecd6", 10);
  g.save(); g.beginPath(); // hull + deck house outline as the clip
  g.moveTo(60, 230); g.lineTo(540, 230); g.lineTo(505, 320); g.lineTo(95, 320); g.closePath();
  g.rect(230, 170, 150, 60); g.rect(268, 110, 34, 60); g.clip();
  g.fillStyle = "#eef0ea"; g.fillRect(0, 0, 600, 420); dzBands(g, 40, 100, 520, 230, 77); g.restore();
  g.strokeStyle = INK; g.lineWidth = 7; g.lineJoin = "round";
  g.beginPath(); g.moveTo(60, 230); g.lineTo(540, 230); g.lineTo(505, 320); g.lineTo(95, 320); g.closePath(); g.stroke();
  g.strokeRect(230, 170, 150, 60); g.strokeRect(268, 110, 34, 60);
  g.lineWidth = 4; g.beginPath(); g.moveTo(150, 230); g.lineTo(150, 80); g.moveTo(460, 230); g.lineTo(460, 80); g.stroke(); // masts
  g.strokeStyle = "#4f7fa8"; g.lineWidth = 5; for (let x = 40; x < 570; x += 60) { g.beginPath(); g.moveTo(x, 345); g.quadraticCurveTo(x + 15, 335, x + 30, 345); g.stroke(); }
  g.strokeStyle = "#7a6a5a"; g.lineWidth = 3; for (const y of [372, 392]) { g.beginPath(); g.moveTo(60, y); g.lineTo(y === 372 ? 360 : 250, y); g.stroke(); } // pencil notes
};
/** a fat arrow pointing right [600x220]; dashed = an outline-only "guess" arrow */
export const fxArrow = (p: Pen, col: string, dashed = false) => {
  const g = p.g, pts: [number, number][] = [[20, 80], [400, 80], [400, 20], [580, 110], [400, 200], [400, 140], [20, 140]];
  if (dashed) {
    g.save(); g.setLineDash([34, 22]); g.lineWidth = 18; g.strokeStyle = col; g.lineJoin = "round";
    g.beginPath(); pts.forEach(([x, y], i) => (i ? g.lineTo(x, y) : g.moveTo(x, y))); g.closePath(); g.stroke(); g.restore();
  } else p.poly(pts, col, 12);
};
/** an aim mark: red ring + cross [300x300] */

  g.beginPath(); g.arc(150, 150, 110, 0, 7); g.stroke();
  g.lineWidth = 16; g.beginPath(); g.moveTo(150, 10); g.lineTo(150, 290); g.moveTo(10, 150); g.lineTo(290, 150); g.stroke();

/** a dashed aim line [1024x64] (lay flat, stretch along) */
export const fxDash = (p: Pen) => {
  const g = p.g; g.fillStyle = "#ffef9a"; for (let x = 0; x < 1024; x += 96) g.fillRect(x, 16, 58, 32);
};
/** torpedo wake: a white foam streak with bubbles, fading toward the start [128x1024] (lay flat) */
export const fxWake = (p: Pen) => {
  const g = p.g, gr = g.createLinearGradient(0, 0, 0, 1024);
  gr.addColorStop(0, "rgba(255,255,255,0.95)"); gr.addColorStop(1, "rgba(255,255,255,0.15)");
  g.fillStyle = gr; g.beginPath(); g.moveTo(52, 0); g.lineTo(76, 0); g.lineTo(104, 1024); g.lineTo(24, 1024); g.closePath(); g.fill();
  g.fillStyle = "rgba(255,255,255,0.85)"; for (let i = 0; i < 60; i++) { const y = (i * 173) % 1024, x = 64 + Math.sin(i * 2.3) * (14 + y / 40); g.beginPath(); g.arc(x, y, 4 + (i % 4), 0, 7); g.fill(); }
};
/** a question mark [200x300] */

};
