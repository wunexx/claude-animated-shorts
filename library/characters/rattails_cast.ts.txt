const RAT = "#8b8078", RAT_D = "#5f554f", RAT_L = "#b8aca2", PINK = "#eaa3a3", PINK_D = "#c97c80", BABY = "#f4bcbc";
const tailPath = (p: Pen, pts: P2[], w0: number, w1: number, col = PINK) => {
  for (let i = 0; i < pts.length - 1; i++) {
    const w = w0 + (w1 - w0) * (i / (pts.length - 1));
    p.stroke([pts[i], pts[i + 1]], w + 7, INK);
  }
  for (let i = 0; i < pts.length - 1; i++) {
    const w = w0 + (w1 - w0) * (i / (pts.length - 1));
    p.stroke([pts[i], pts[i + 1]], w, col);
  }
  for (let i = 2; i < pts.length - 1; i += 2) p.dot(pts[i][0], pts[i][1], 1.6, PINK_D);
};
/** a rat [400x300 side / sit; 300x400 hang]. pose run (side view facing +x, ph = gait 0..1) | sit (front, upright) |
 *  hang (held by the scruff, dangling). tail = length 0..1 (0 = a stub with a pink nub). baby = a hairless pink pup. */
export const fxRat = (p: Pen, pose: "run" | "sit" | "hang", tail: number, ph = 0, baby = false, expr: "calm" | "shock" | "smug" = "calm") => {
  const fur = baby ? BABY : RAT, furD = baby ? PINK_D : RAT_D, belly = baby ? "#f8d0cf" : RAT_L;
  const eye = (x: number, y: number, r: number) => {
    if (baby) { p.arc(x, y, r * 1.2, r * 0.6, 20, 160, 3, INK); return; }
    if (expr === "shock") { p.ell(x, y, r * 1.3, r * 1.5, "#ffffff", 3); p.dot(x, y, r * 0.55, INK); }
    else if (expr === "smug") { p.ell(x, y, r, r, INK, 0); p.poly([[x - r * 1.3, y - r * 0.2], [x + r * 1.3, y - r * 0.2], [x + r * 1.3, y - r * 1.6], [x - r * 1.3, y - r * 1.6]], fur, 0); }
    else p.dot(x, y, r, INK);
  };
  if (pose === "run") {
    const s = Math.sin(ph * Math.PI * 2), c = Math.cos(ph * Math.PI * 2);
    if (tail > 0) {
      const L = 150 * tail, pts: P2[] = [];
      for (let i = 0; i <= 12; i++) { const k = i / 12; pts.push([112 - L * k, 198 - 40 * Math.sin(k * 3 + ph * 6) * k - 10 * k]); }
      tailPath(p, pts, 13, 3);
    } else { p.ell(104, 196, 12, 9, PINK, 4); p.ell(98, 196, 5, 5, "#d95a5a", 0); }
    // legs: back pair / front pair alternate
    for (const [lx, a] of [[140, s], [250, -s]] as [number, number][]) {
      p.ell(lx + 22 * a, 248 - 8 * Math.max(0, c * (lx > 200 ? -1 : 1)), 20, 12, furD, 4);
      p.ell(lx - 18 * a, 244, 17, 11, fur, 4);
    }
    p.ell(185, 192 - 5 * Math.abs(s), 96, 58, fur, 6);
    p.clip([[95, 200], [280, 200], [280, 260], [95, 260]], () => p.ell(190, 200, 86, 42, belly, 0));
    p.ell(185, 192 - 5 * Math.abs(s), 96, 58, null, 6);
    const hy = 168 - 4 * Math.abs(s);
    p.poly([[262, hy - 38], [318, hy - 22], [372, hy + 12], [318, hy + 38], [262, hy + 36]], fur, 6);
    p.ell(268, hy, 48, 40, fur, 0);
    p.arc(268, hy, 48, 40, 120, 250, 6, INK);
    p.ell(258, hy - 44, 24, 28, baby ? BABY : PINK, 5); p.ell(258, hy - 42, 13, 16, PINK_D, 0);
    p.dot(372, hy + 12, 9, "#d95a5a");
    eye(318, hy - 6, 7);
    for (const dy of [-6, 4, 14]) p.stroke([[350, hy + 10], [400, hy + dy]], 2, INK);
    return;
  }
  if (pose === "sit") {
    if (tail > 0) {
      const L = 170 * tail, pts: P2[] = [];
      for (let i = 0; i <= 12; i++) { const k = i / 12; pts.push([240 + L * 0.6 * k, 270 - L * 0.55 * Math.sin(k * 2.2)]); }
      tailPath(p, pts, 13, 3);
    } else { p.ell(255, 262, 12, 9, PINK, 4); p.ell(262, 262, 5, 5, "#d95a5a", 0); }
    p.ell(200, 205, 82, 82, fur, 6);
    p.ell(200, 222, 52, 58, belly, 0);
    p.ell(160, 285, 26, 13, furD, 4); p.ell(240, 285, 26, 13, furD, 4);
    p.ell(200, 118, 64, 56, fur, 6);
    p.ell(146, 72, 28, 32, baby ? BABY : PINK, 5); p.ell(254, 72, 28, 32, baby ? BABY : PINK, 5);
    p.ell(146, 74, 15, 18, PINK_D, 0); p.ell(254, 74, 15, 18, PINK_D, 0);
    p.ell(200, 140, 30, 22, belly, 0);
    p.dot(200, 132, 10, "#d95a5a");
    eye(176, 108, 8); eye(224, 108, 8);
    for (const sx of [-1, 1]) for (const dy of [-6, 6]) p.stroke([[200 + sx * 22, 142], [200 + sx * 70, 140 + dy * 1.5]], 2, INK);
    // front paws up by the chest
    p.ell(176, 182, 15, 12, PINK, 4); p.ell(224, 182, 15, 12, PINK, 4);
    return;
  }
  // hang: held up by the scruff (canvas 300 x 400): head at top, body dangling, legs splayed
  const sw = Math.sin(ph * Math.PI * 2) * 6;
  if (tail > 0) {
    const L = 170 * tail, pts: P2[] = [];
    for (let i = 0; i <= 12; i++) { const k = i / 12; pts.push([150 + sw + 30 * Math.sin(k * 4 + ph * 6) * k, 300 + L * k]); }
    tailPath(p, pts, 13, 3);
  } else { p.ell(150 + sw, 304, 10, 12, PINK, 4); p.ell(150 + sw, 312, 5, 5, "#d95a5a", 0); }
  for (const sx of [-1, 1]) { p.ell(150 + sx * 52 + sw, 270, 13, 20, PINK, 4); p.ell(150 + sx * 48, 170, 13, 18, PINK, 4); }
  p.ell(150 + sw * 0.6, 215, 56, 88, fur, 6);
  p.ell(150 + sw * 0.6, 230, 34, 60, belly, 0);
  p.ell(150, 96, 52, 46, fur, 6);
  p.ell(106, 58, 22, 26, PINK, 5); p.ell(194, 58, 22, 26, PINK, 5);
  p.ell(150, 120, 26, 20, belly, 0); p.dot(150, 130, 8, "#d95a5a");
  eye(130, 92, 7); eye(170, 92, 7);
  p.arc(150, 112, 10, 6, 20, 160, 3, INK);
};

/** iron scissors [240x240], open 0..1 */

  for (const sg of [-1, 1]) {

    const pt = (x: number, y: number): P2 => [120 + x * c - y * s, 150 + x * s + y * c];

  }
  p.dot(120, 150, 7, "#8a8d94");

/** a weed [300x420]: stage 0 growing in soil (leaves + yellow flower), 1 pulled out with roots dangling */
