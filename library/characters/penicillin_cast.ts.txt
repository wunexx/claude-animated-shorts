/** stack of glass dishes seen from the side (400 x 700): n dishes, lean = wobble angle (deg) */
export const fxStack = (p: Pen, n: number, lean = 0) => {
  const dh = 600 / n;
  for (let i = 0; i < n; i++) {
    const y = 680 - (i + 1) * dh, dx = Math.sin(i * 1.7) * 14 + lean * i * 1.1, cx = 200 + dx, w = 150 - (i % 3) * 6;
    p.poly([[cx - w, y + dh * 0.15], [cx + w, y + dh * 0.15], [cx + w, y + dh * 0.95], [cx - w, y + dh * 0.95]], "#d7ecf0", 7);

    p.stroke([[cx - w + 14, y + dh * 0.25], [cx - w + 60, y + dh * 0.25]], 5, "rgba(255,255,255,0.9)");
  }
};
/** a petri dish top-down (600 x 600): mould m 0..1 (fuzzy blue-green colony), clear ring c 0..1 (germs near it vanish) */
export const fxDish = (p: Pen, m: number, c: number) => {
  const C = 300, R = 270;
  p.ell(C, C, R + 14, R + 14, "#dcecf2", 12);
  p.ell(C, C, R - 6, R - 6, "#f0cf7a", 6);
  const mx = C - 40, my = C - 20, ringR = 70 + 120 * m;
  if (c > 0) p.ell(mx, my, ringR * (0.6 + 0.4 * c), ringR * (0.6 + 0.4 * c), mix("#f0cf7a", "#fbf0cc", c), 0);
  for (let i = 0; i < 120; i++) {
    const a = hrand(i) * Math.PI * 2, d = Math.sqrt(hrand(i + 500)) * (R - 30), x = C + Math.cos(a) * d, y = C + Math.sin(a) * d;
    const dm = Math.hypot(x - mx, y - my);
    let s = 6 + hrand(i + 900) * 9;
    if (m > 0 && dm < ringR) s *= 1 - c;
    if (m > 0 && dm < 70 * m + 10) continue;
    if (s > 1) p.ell(x, y, s, s, ["#f4b02e", "#fff1b8", "#e8962a"][i % 3], 3);
  }
  if (m > 0) {
    const rr = 70 * m;
    for (let i = 0; i < 18; i++) { const a = (i / 18) * Math.PI * 2; p.ell(mx + Math.cos(a) * rr * 0.8, my + Math.sin(a) * rr * 0.8, rr * 0.36, rr * 0.36, "#f3f7ee", 0); }
    p.ell(mx, my, rr, rr * 0.95, "#3f9a86", 6);
    p.ell(mx - rr * 0.2, my - rr * 0.2, rr * 0.55, rr * 0.5, "#5cb59a", 0);
    for (let i = 0; i < 26; i++) p.dot(mx + (hrand(i + 50) - 0.5) * rr * 1.6, my + (hrand(i + 80) - 0.5) * rr * 1.6, 5, "#2f7a6b");
  }
  p.arc(C, C, R - 30, R - 30, 200, 250, 14, "rgba(255,255,255,0.9)");
};

/** a white mouse (200 x 140) at (x, y): mode "sick" | "dead" | "alive" */
const mouse = (p: Pen, x: number, y: number, mode: string, s = 1) => {
  const up = mode === "dead";
  p.stroke([[x + 60 * s, y + 10 * s], [x + 95 * s, y - 5 * s], [x + 110 * s, y + 20 * s]], 6 * s, "#e7a4a8");
  p.ell(x, y, 62 * s, up ? 34 * s : 40 * s, "#f6f3f0", 7 * s);
  p.ell(x - 52 * s, y - 34 * s, 20 * s, 20 * s, "#f6f3f0", 6 * s); p.ell(x - 52 * s, y - 34 * s, 10 * s, 10 * s, "#efa6b0", 0);
  p.ell(x - 22 * s, y - 40 * s, 20 * s, 20 * s, "#f6f3f0", 6 * s); p.ell(x - 22 * s, y - 40 * s, 10 * s, 10 * s, "#efa6b0", 0);
  p.dot(x - 64 * s, y + 4 * s, 7 * s, "#e7848e");
  if (up) {
    for (const ex of [-44, -18]) { p.stroke([[x + (ex - 7) * s, y - 15 * s], [x + (ex + 7) * s, y - 1 * s]], 5 * s); p.stroke([[x + (ex + 7) * s, y - 15 * s], [x + (ex - 7) * s, y - 1 * s]], 5 * s); }
    for (const lx of [-20, 10, 30]) p.stroke([[x + lx * s, y - 30 * s], [x + (lx - 6) * s, y - 62 * s]], 5 * s, "#e7a4a8"); // belly-up: legs in the air
  } else if (mode === "sick") {
    for (const ex of [-44, -18]) p.stroke([[x + (ex - 7) * s, y - 8 * s], [x + (ex + 7) * s, y - 8 * s]], 5 * s);
  } else {
    for (const ex of [-44, -18]) p.dot(x + ex * s, y - 10 * s, 6 * s);
    p.arc(x - 31 * s, y + 6 * s, 10 * s, 7 * s, 20, 160, 4 * s, INK);
  }
};

/** a bacterium (520 x 520): wall 0..1 (brick ring built round it), crack 0..1, mood */
export const fxBug = (p: Pen, wall: number, crack: number, mood: string, seed = 1) => {
  const C = 260, r = 150;
  if (crack >= 1) return;
  p.ell(C, C, r, r * 0.92, mood === "worried" ? "#f2b25a" : "#f4a93a", 8);
  p.ell(C - 40, C - 40, r * 0.45, r * 0.32, "#f8c86a", 0);
  for (let i = 0; i < 5; i++) p.dot(C + (hrand(i + seed * 10) - 0.5) * 160, C + 40 + (hrand(i + seed * 20) - 0.5) * 70, 8, "#d07f22");
  for (const ex of [-45, 45]) {
    if (mood === "worried") { p.ell(C + ex, C - 20, 20, 24, WHITE, 6); p.dot(C + ex, C - 14, 9); p.stroke([[C + ex - 22, C - 58 + (ex > 0 ? 8 : -2)], [C + ex + 22, C - 58 + (ex > 0 ? -2 : 8)]], 7); }
    else { p.ell(C + ex, C - 20, 20, 24, WHITE, 6); p.dot(C + ex + 4, C - 16, 9); }
  }
  if (mood === "worried") p.ell(C, C + 48, 18, 22, "#7a2a3a", 6); else p.arc(C, C + 30, 46, 30, 20, 160, 8, INK);
  // brick wall: segments around the cell, built counter-clockwise from the bottom
  const n = 22, built = Math.floor(n * wall);
  for (let i = 0; i < built; i++) {
    const a0 = Math.PI / 2 + (i / n) * Math.PI * 2, a1 = a0 + (Math.PI * 2) / n * 0.9;
    const off = crack > 0 ? crack * 120 * (0.5 + hrand(i)) : 0, oa = (a0 + a1) / 2;
    const ox = Math.cos(oa) * off, oy = Math.sin(oa) * off, R0 = r + 18, R1 = r + 62;
    p.poly([[C + ox + Math.cos(a0) * R0, C + oy + Math.sin(a0) * R0], [C + ox + Math.cos(a0) * R1, C + oy + Math.sin(a0) * R1],
      [C + ox + Math.cos(a1) * R1, C + oy + Math.sin(a1) * R1], [C + ox + Math.cos(a1) * R0, C + oy + Math.sin(a1) * R0]], i % 2 ? "#c0503a" : "#b04632", 6);
  }
};
