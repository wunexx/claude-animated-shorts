/** King Canute: crown, ginger beard, red robe with a gold sash (bare = no crown, after he gives it up) */
export const canuteO = (bare = false, wet = false): Outfit => ({ jacket: wet ? "#8a1d26" : "#b3282d", pants: "#8e1f2a", boots: "#5a3a22", hat: bare ? "hair" : "crown", cap: "#f2c230",

/** Anglo-Danish courtiers in long robes (v 0..3) */
export const courtierO = (v: number): Outfit => {
  const J = ["#3c7a4e", "#3d5f9e", "#7a4a8e", "#c08a2a"][v % 4], S = ["#e8c75a", "#e9e2cf", "#e8c75a", "#7a2a2a"][v % 4];
  const H = ["#5a3a22", "#2e2420", "#c9a060", "#6e4a2a"][v % 4];
  return { jacket: J, pants: shade(J, 0.7), boots: "#4a3222", hat: "hair", hair: H, dress: J, sash: S, moustache: v % 2 ? H : undefined, beard: v === 2 ? H : undefined };
};

/** carved oak throne with a red cushion and a TALL arched back that rises above a seated king's crown [500 x 900];
 *  seat top at y ~ 610 (0.6 m on a 1.86 m card). Keep the back's top edge above the sitter's head: an edge at face height
 *  beside the head reads as a cut through the face (Canute 2026-11-20). */
export const fxThrone = (p: Pen) => {
  const oak = "#8a5530", dk = "#5e3a22", gold = "#f2c230";
  const back: P2[] = [[110, 670], [110, 120], ...curve([110, 120], [250, -10], [390, 120], 16).slice(1), [390, 670]];
  p.poly(back, oak, 12);
  p.poly(rrect(150, 150, 350, 620, 22), "#b3282d", 9);
  for (let i = 0; i < 4; i++) p.stroke([[170 + i * 53, 170], [170 + i * 53, 600]], 5, mul("#b3282d", [0.8, 0.8, 0.8]));
  for (const x of [110, 390]) { p.poly(rrect(x - 26, 110, x + 26, 880, 12), dk, 11); p.ell(x, 104, 34, 30, gold, 10); }
  p.poly(rrect(70, 610, 430, 670, 18), oak, 11);
  p.poly(rrect(84, 592, 416, 628, 16), "#c8403a", 9);
  for (const x of [96, 404]) p.poly(rrect(x - 18, 666, x + 18, 890, 8), dk, 10);
  p.poly(rrect(96, 760, 404, 784, 8), dk, 8);
  p.ell(250, 60, 30, 30, gold, 9);
};

/** a curling wave with a foam crest [900 x 500]; k 0..1 = how far it has broken (curl -> flat foam) */
export const fxWave = (p: Pen, k: number) => {
  const deep = "#3f9fd6", lite = "#8fd3f4", foam = "#f4fbff", h = 300 * (1 - 0.6 * k);
  const top: P2[] = [];
  for (let i = 0; i <= 30; i++) { const x = 30 + (840 * i) / 30, e = Math.sin((Math.PI * i) / 30); top.push([x, 480 - h * (0.25 + 0.75 * e) - 14 * Math.sin(i * 1.1 + k * 4)]); }
  p.poly([[30, 490], ...top, [870, 490]], deep, 12);
  p.poly([[60, 490], ...top.slice(2, 29).map(([x, y]) => [x, y + 0.5 * (490 - y)] as P2), [840, 490]], lite, 0);
  p.stroke(top.slice(3, 28), 46 * (1 - 0.4 * k), INK);
  p.stroke(top.slice(3, 28), 34 * (1 - 0.4 * k), foam);
  // the curl at the crest
  const [cx, cy] = top[15];
  if (k < 0.8) p.ell(cx - 30, cy + 26, 56 * (1 - k), 40 * (1 - k), foam, 10, 170, 470);
  for (let i = 0; i < 7; i++) p.stroke([[110 + i * 110, 470], [150 + i * 110, 462]], 8, foam);
};

/** wooden cross on a stone base with Canute's gold crown hanging on it [500 x 900] */
export const fxCrossCrown = (p: Pen) => {
  const wood = "#8a5530", gold = "#f2c230";
  p.poly([[140, 890], [360, 890], [330, 800], [170, 800]], "#9c9488", 11);
  p.poly(rrect(222, 60, 278, 810, 10), wood, 12);
  p.poly(rrect(70, 220, 430, 276, 10), wood, 12);
  for (let i = 0; i < 6; i++) p.stroke([[236, 110 + i * 120], [236, 170 + i * 120]], 4, mul(wood, [0.7, 0.7, 0.7]));
  // the crown hangs from the right arm on a cord
  p.stroke([[370, 276], [370, 330]], 6, INK);
  const yb = 470, w = 82, h = 120, sp: P2[] = [[370 - w, yb]];
  for (let i = 0; i <= 4; i++) { const x = 370 - w + (2 * w * i) / 4; sp.push([x, yb - h]); if (i < 4) sp.push([x + w / 4, yb - h * 0.5]); }
  sp.push([370 + w, yb]);
  p.shadedFill(sp, gold, mul(gold, [0.75, 0.6, 0.4]), -Math.PI / 4, w, 0.4);
  p.poly(sp, null, 10);
  p.poly(rrect(370 - w, yb - 26, 370 + w, yb + 8, 6), gold, 9);
  for (let i = 0; i <= 4; i++) p.dot(370 - w + (2 * w * i) / 4, yb - h - 8, 10, i % 2 ? "#c0353a" : "#3a8fd0");
  p.dot(370, yb - 10, 12, "#c0353a");
};

/** alarm-clock bits for the moon [700 x 700]: two bells, hammer, legs, ringing lines; ring 0..1 shakes */
export const fxClockBits = (p: Pen, ring: number) => {
  const gold = "#f2c230", sh = Math.sin(ring * 40) * 14 * ring;
  for (const s of [-1, 1]) {
    const cx = 350 + s * 190 + sh, cy = 150;
    p.ell(cx, cy, 90, 70, gold, 12, 180, 360, s * 30);
    p.stroke([[cx - 70, cy + 4], [cx + 70, cy + 4]], 14, INK);
    p.stroke([[350 + s * 120, 230], [350 + s * 165, 175]], 14, INK);
    p.stroke([[350 + s * 140, 560], [350 + s * 210, 670]], 18, INK);
    p.ell(350 + s * 214, 672, 34, 16, INK, 0);
  }
  p.stroke([[350 + sh * 2, 110], [350 + sh * 2, 190]], 12, INK);
  p.ell(350 + sh * 2, 96, 20, 20, INK, 0);
  if (ring > 0.05) for (const s of [-1, 1]) for (let i = 0; i < 3; i++) {
    const a = rad(-60 - i * 30) * s, r = 260 + 10 * i;
    p.stroke([[350 + s * Math.abs(Math.cos(a)) * r, 150 - Math.abs(Math.sin(a)) * r * 0.6], [350 + s * Math.abs(Math.cos(a)) * (r + 70), 150 - Math.abs(Math.sin(a)) * (r + 70) * 0.6]], 12, "#fff3b0");
  }
};

/** a splash of sea water over someone's boots [800 x 800]: a foam band + flying teardrop drops; k 0..1 */
export const fxSpray = (p: Pen, k: number) => {
  const foam = "#f4fbff", blue = "#8fd3f4", rise = Math.sin(Math.min(1, k * 1.4) * Math.PI);
  p.poly([[60, 760], ...Array.from({ length: 13 }, (_, i) => [60 + i * 56.7, 700 - (i % 2 ? 70 : 30) * (0.5 + rise)] as P2), [740, 760]], foam, 11);
  for (let i = 0; i < 9; i++) {
    const a = rad(-160 + i * 17.5), d = 120 + 330 * k, x = 400 + d * Math.cos(a), y = 680 + d * Math.sin(a) * 1.2 + 420 * k * k, r = 30 * (1 - 0.6 * k);
    if (y > 40 && y < 720) { p.ell(x, y, r, r * 1.15, i % 2 ? blue : foam, 8); p.ell(x - r * 0.3, y - r * 0.35, r * 0.25, r * 0.25, '#ffffff', 0); }
  }
};
