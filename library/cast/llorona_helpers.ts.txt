/** La Llorona: an unlit, faintly glowing card (base = the hem, ~0.15 m over the water), with a soft halo behind */
const ghostAt = (out: Stage, cam: CamInfo, t: number, base: V3, o: GhostOpts, scale = 1, glow = 1.15) => {
  const h = 2.16 * scale, w = 1.296 * scale, bob = 0.07 * Math.sin(t * 1.7) * scale;
  const cx = base[0], cy = base[1] + bob + 0.47 * h, cz = base[2];
  const d = cam.pos.distanceTo(tmp.set(cx, cy, cz));
  const [cw, ch, k] = tierOf(d, h);
  const phase = q((t * 0.45) % 1, 24), lid = o.expr === "weep" || o.expr === "wail" ? 0 : lidAt(t, 501);
  const go: GhostOpts = { ...o, phase, blink: lid };
  out.fx.push({ key: `ghost|${fkey(go)}`, cw, ch, draw: (p) => { p.g.scale(k, k); ghost(p, go); }, x: cx, y: cy, z: cz, w, h, glow, see: true });
  if (base[1] < 2.5) { // she stands in the water: rings spread from her hem
    const ru = q((t * 0.5) % 1, 30);
    out.fx.push({ key: `ring${ru}`, cw: 600, ch: 600, draw: (p) => fxRing(p, ru), x: cx, y: 0.04, z: cz, w: 2.6 * scale, h: 2.6 * scale, flat: true, soft: true });
  }
  // halo a little behind the card (away from the lens) so it never z-fights the drawing
  const back = tmp.set(cx - cam.pos.x, 0, cz - cam.pos.z).normalize().multiplyScalar(0.25 * scale);
  out.fx.push({ key: "glow", cw: 256, ch: 256, draw: (p) => fxGlow(p), x: cx + back.x, y: cy - 0.1 * h, z: cz + back.z, w: 2.6 * scale, h: 2.9 * scale, soft: true });
};

/** sound arcs around a point: both sides, opening outward; s = size (m), a = loudness 0..1 */
const wail = (out: Stage, cam: CamInfo, t: number, at: V3, s: number, a = 1, n = 3) => {
  const u = q((t * 1.4) % 1, 20), al = q(a, 10), R = cam.right;
  for (const side of [1, -1]) out.fx.push({ key: `wail|${u}|${n}|${al}`, cw: 400, ch: 400, draw: (p) => fxWail(p, u, n, al),
    x: at[0] + side * R.x * 0.5 * s, y: at[1] + side * R.y * 0.5 * s, z: at[2] + side * R.z * 0.5 * s, w: s, h: s, rot: side > 0 ? 0 : Math.PI, soft: true, glow: 1.3 });
};

/** the cry arriving at a listener from the screen side `from` (-1 left / +1 right): arcs beside the head, not on it */
const arrive = (out: Stage, cam: CamInfo, t: number, head: V3, s: number, from = -1, a = 0.6) => {
  const u = q((t * 1.4) % 1, 20), al = q(a, 10), R = cam.right, off = from * (0.55 + 0.5 * s);
  out.fx.push({ key: `wail|${u}|2|${al}`, cw: 400, ch: 400, draw: (p) => fxWail(p, u, 2, al),
    x: head[0] + R.x * off, y: head[1] + R.y * off, z: head[2] + R.z * off, w: s, h: s, rot: from < 0 ? 0 : Math.PI, soft: true, glow: 1.3 });
};

const flame = (out: Stage, t: number, at: V3, s = 0.5, i = 0) => {
  const u = q((t * 2.3 + hh(i, 3)) % 1, 12);
  out.fx.push({ key: `flame|${u}`, cw: 200, ch: 300, draw: (p) => fxFlame(p, u), x: at[0], y: at[1] + 0.75 * s, z: at[2], w: s, h: 1.5 * s, glow: 2.6 });
};

// ---- the lake city ----
const roofPeople = (out: Stage, cam: CamInfo, t: number, n: number, shockAt: number) => {
  const hs = lakeHouses().filter((h) => h.z > 0 && Math.abs(h.x) < 60).sort((a, b) =>
    Math.hypot(LX + a.x - cam.pos.x, a.z - cam.pos.z) - Math.hypot(LX + b.x - cam.pos.x, b.z - cam.pos.z)).slice(0, n);
  hs.forEach((h, i) => {
    const pop = smooth((t - shockAt - 0.12 * (i % 5)) / 0.25);
    if (pop <= 0) return;
    const y = ISLE + h.h + 0.05 - (1 - pop) * 1.6; // they pop up over the parapet
    front(out, cam, t, 200 + i, aztec(i), `az${i % 6}`, [LX + h.x + (hh(i, 4) - 0.5) * 2, y, h.z + h.d / 2 - 1.2], { mode: "stand", expr: i % 3 ? "shock" : "nervous", look: i % 2 ? -1 : 1 }, { blob: false });
  });
};

/** the sound pulse path of the diagram: from the ghost up to the lid and bent back down to the fisherman's ear */
export const pulsePath = (u: number): V3 => {
  const a: V3 = [G2[0], G2[1] + 2.2, G2[2]], b: V3 = [C2[0], C2[1] + 1.6, C2[2]];
  const top = LID_Y - 1.2, apex = 0.45;
  const y = u < apex ? a[1] + (top - a[1]) * Math.sin((u / apex) * Math.PI / 2) : b[1] + (top - b[1]) * Math.cos(((u - apex) / (1 - apex)) * Math.PI / 2);
  return [a[0] + (b[0] - a[0]) * u, y, a[2] + (b[2] - a[2]) * u];
};
const PULSE0 = 49.62, PULSE_T = 1.45;
const pulses = (out: Stage, cam: CamInfo, t: number, n = 3) => {
  const up = cam.up ?? tmp.set(0, 1, 0);
  for (let k = 0; k < n; k++) {
    const u = (t - PULSE0 - 0.32 * k) / PULSE_T;
    if (u <= 0 || u >= 1) continue;
    const p = pulsePath(u), p2 = pulsePath(Math.min(1, u + 0.02));
    const d = [p2[0] - p[0], p2[1] - p[1], p2[2] - p[2]];
    const sx = d[0] * cam.right.x + d[1] * cam.right.y + d[2] * cam.right.z, sy = d[0] * up.x + d[1] * up.y + d[2] * up.z;
    const dc = Math.hypot(p[0] - cam.pos.x, p[1] - cam.pos.y, p[2] - cam.pos.z), s = 7.5 * Math.min(1.2, Math.max(0.3, dc / 45));
    out.fx.push({ key: "pulse", cw: 300, ch: 300, draw: (pp) => fxPulse(pp), x: p[0], y: p[1], z: p[2], w: s, h: s, rot: Math.atan2(sy, sx), soft: true, glow: 1.6 });
  }
  // the bounce flash where the pulses meet the lid
  const tb = t - (PULSE0 + 0.45 * PULSE_T);
  if (tb >= 0 && tb < 0.3) {
    const u = q(tb / 0.3, 20), p = pulsePath(0.45);
    out.fx.push({ key: `burstW${u}`, cw: 500, ch: 500, draw: (pp) => fxBurst(pp, u, "#fff1c4", 8, 30, 200), x: p[0], y: p[1] + 0.6, z: p[2], w: 3.4, h: 3.4 });
  }
};
/** the cold air layer just over the water (mist banks fading in on "chills") */
const coldMist = (out: Stage, t: number, a: number) => {
  if (a <= 0.02) return;
  const al = q(a * 0.32, 160);
  for (let i = 0; i < 9; i++) {
    const u = i / 8, x = G2[0] + (C2[0] - G2[0]) * u + (hh(i, 1) - 0.5) * 14, z = G2[2] + (C2[2] - G2[2]) * u - 4;
    out.fx.push({ key: `mist|${i % 3}|${al}`, cw: 512, ch: 128, draw: (p) => fxMist(p, i % 3, "160,195,250", al), x: x + 0.6 * Math.sin(t * 0.3 + i), y: 0.7, z, w: 20, h: 3.2, soft: true });
  }
};

// ---- the bathroom ----
const bathroom = (out: Stage, cam: CamInfo, t: number) => {
  const kid: V3 = [BATH[0] - 0.1, 0.5, BATH[2] - 0.75];
  front(out, cam, t, 41, { ...kidO(3), hat: "hair" }, "bathkid", kid, { mode: "sit", expr: "sing", headTilt: -6 }, { blob: false, scale: 0.8 });
  out.fx.push({ key: "tub", cw: 700, ch: 300, draw: fxTub, x: BATH[0] - 0.1, y: 0.36, z: BATH[2] - 0.45, w: 1.6, h: 0.69 });
  // notes rise to the ceiling (2.5 m), bounce, and come back down bigger
  for (let k = 0; k < 6; k++) {
    const u = ((t - 51.5 + k * 0.33) % 1.9) / 1.9;
    if (t < 51.5 - k * 0.33 + 0.0 || u < 0) continue;
    const up = u < 0.5, v = up ? u / 0.5 : (u - 0.5) / 0.5;
    const y = up ? 1.25 + (2.42 - 1.25) * Math.sin(v * Math.PI / 2) : 2.42 - (2.42 - 1.05) * (1 - Math.cos(v * Math.PI / 2));
    const x = BATH[0] - 0.1 + (up ? -0.15 - 0.25 * v : -0.4 + 0.9 * v) + 0.06 * Math.sin(k), s = up ? 0.18 : 0.18 + 0.12 * v;
    out.fx.push({ key: `note${k % 2}`, cw: 220, ch: 260, draw: (p) => fxNote(p, k % 2, "#2b3a6e"), x, y, z: BATH[2] - 0.45 + 0.05 * k, w: s, h: s * 1.18, rot: 0.3 * Math.sin(u * 8 + k) });
  }
};
