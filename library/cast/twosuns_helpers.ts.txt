// (sunbeams from a camera-relative sun). Paste below "// ---- library staging helpers" in cast.ts and add to the toon
// import: dog, DogOpts, fxSweat, fxSteam, fxBeam, fxStar, fxOrbit, fxHeat. starField/beams use SP / L from story.ts
// (edit those two to your own space / sea origin). Needs library/characters/twosuns_cast.ts.txt. Origin: Two Suns
/** the dog (side view card); mirror = facing screen left */
const dogAt = (out: Stage, cam: CamInfo, t: number, p: V3, o: DogOpts, mirror = false, shadow = true) => {
  const d = cam.pos.distanceTo(tmp.set(p[0], p[1] + 0.4, p[2]));
  const [w, h, k] = tierOf(d, 0.9);
  const o2: DogOpts = { ...o, blink: lidAt(t, 90) };
  out.places.push({ key: `dog|${fkey(o2)}`, x: p[0], y: p[1], z: p[2], mirror, draw: (pp) => dog(pp, o2), size: [w, h, k] });
  if (shadow) blob(out, p[0], p[1], p[2], 1.05);
};
/** sweat drops sliding down beside a head (world point), both sides */
const sweat = (out: Stage, cam: CamInfo, t: number, head: V3, s = 1, i = 0) => {
  for (const side of [-1, 1]) {
    const u = (t * 0.9 + hh(i, side + 5)) % 1;
    if (u > 0.85) continue;
    const off = side * 0.2 * s, dy = -0.18 * s * u;
    out.fx.push({ key: "sweat", cw: 120, ch: 160, draw: fxSweat, x: head[0] + cam.right.x * off, y: head[1] + 0.05 * s + dy, z: head[2] + cam.right.z * off, w: 0.07 * s, h: 0.093 * s });
  }
};
/** rising steam: n columns spread over a box around c (dx, dz half sizes), column height size, strength a */
const steam = (out: Stage, t: number, c: V3, dx: number, dz: number, n: number, size: number, a: number, seed = 0) => {
  if (a <= 0.02) return;
  for (let i = 0; i < n; i++) {
    const u = (t * 0.32 + hh(i, seed + 1)) % 1;
    const al = q(a * Math.sin(Math.PI * u), 12);
    if (al <= 0) continue;
    const s = size * (0.75 + 0.5 * hh(i, seed + 2));
    out.fx.push({ key: `steam|${i % 4}|${al}`, cw: 256, ch: 512, draw: (p) => fxSteam(p, i % 4, al), x: c[0] + (hh(i, seed + 3) - 0.5) * 2 * dx, y: c[1] + s * (0.45 + 0.35 * u), z: c[2] + (hh(i, seed + 4) - 0.5) * 2 * dz, w: s * 0.5, h: s, soft: true });
  }
};
const heat = (out: Stage, t: number, p: V3, s: number, i = 0) => {
  const u = q((t * 0.8 + hh(i, 7)) % 1, 16);
  out.fx.push({ key: `heat|${u}`, cw: 256, ch: 512, draw: (pp) => fxHeat(pp, u), x: p[0], y: p[1] + s * 0.5, z: p[2], w: s * 0.5, h: s, soft: true });
};
const star = (out: Stage, p: V3, s: number, col: string, glow = 2.6) =>
  out.fx.push({ key: `star|${col}`, cw: 256, ch: 256, draw: (pp) => fxStar(pp, col), x: p[0], y: p[1], z: p[2], w: s, h: s, soft: true, glow });

const STAR_COLS = ["255,236,200", "200,220,255", "255,210,170", "255,250,235"];
/** shot 13: a field of star systems, a bit over half of them pairs circling each other */
const starField = (out: Stage, t: number) => {
  for (let i = 0; i < 11; i++) {
    const z = -340 - hh(i, 1) * 700, dd = -200 - z, x = (((i * 0.7549 + 0.3) % 1) - 0.5) * 0.44 * dd + 10, y = (((i * 0.5698 + 0.15) % 1) - 0.5) * 0.8 * dd + 6;
    const c: V3 = SP(x, y, z), d = -200 - z, s = d * (0.07 + 0.02 * hh(i, 4));
    const col = STAR_COLS[i % 4], col2 = STAR_COLS[(i + 1) % 4];
    if (i % 4 !== 3) {
      out.fx.push({ key: "orbit", cw: 512, ch: 512, draw: fxOrbit, x: c[0], y: c[1], z: c[2] + 0.5, w: s * 2.2, h: s * 2.2 * (0.35 + 0.4 * hh(i, 8)), soft: true, glow: 1.2 });
      const r = s * 0.55, a = t * (0.9 + 0.8 * hh(i, 6)) + hh(i, 7) * 6.28, tilt = 0.35 + 0.4 * hh(i, 8);
      star(out, [c[0] + r * Math.cos(a), c[1] + r * Math.sin(a) * tilt, c[2] + r * Math.sin(a) * 0.3], s, col);
      star(out, [c[0] - r * 0.8 * Math.cos(a), c[1] - r * 0.8 * Math.sin(a) * tilt, c[2] - r * 0.8 * Math.sin(a) * 0.3], s * 0.8, col2);
    } else star(out, c, s * 1.05, col);
  }
};
/** shot 9: sunbeams from one of the two suns down onto the sea (camera looks along +z; az from DISCS) */
const beams = (out: Stage, cam: CamInfo, t: number, az: number, el: number, k: number, seed: number) => {
  if (k <= 0.02) return;
  const a = (az * Math.PI) / 180, e = (el * Math.PI) / 180, fw = [0, 1];
  const hx = fw[0] * Math.cos(a) - fw[1] * Math.sin(a), hz = fw[1] * Math.cos(a) + fw[0] * Math.sin(a);
  const d = [hx * Math.cos(e), Math.sin(e), hz * Math.cos(e)];
  const up = cam.up ?? tmp.set(0, 1, 0), sx = d[0] * cam.right.x + d[1] * cam.right.y + d[2] * cam.right.z, sy = d[0] * up.x + d[1] * up.y + d[2] * up.z;
  const rot = Math.atan2(-sx, sy), len = 70, al = q(k * 0.6, 10);
  for (let i = 0; i < 3; i++) {
    const tx = L(-24 + 24 * i + 6 * hh(i, seed), 0, 46 + 20 * hh(i, seed + 1));
    out.fx.push({ key: `beam|${al}`, cw: 128, ch: 1024, draw: (p) => { p.g.globalAlpha = al; fxBeam(p); }, x: tx[0] + d[0] * len * 0.5, y: d[1] * len * 0.5, z: tx[2] + d[2] * len * 0.5, w: 7, h: len, rot, soft: true, glow: 1.35 });
  }
};
